## Exercise 1: CRUD API for a Task Manager
**Difficulty:** ★★★☆☆  
**Concepts:** Flask app setup, route decorators, HTTP methods, JSON request/response  

**Description:**  
Build a RESTful API for managing tasks. Implement full CRUD operations (Create, Read, Update, Delete) using Flask routes with appropriate HTTP methods. Store tasks in memory using a Python list/dictionary.

**Example Starter Code:**
```python
from flask import Flask, request, jsonify

app = Flask(__name__)

# In-memory task storage
tasks = [
    {"id": 1, "title": "Learn Flask", "completed": False},
    {"id": 2, "title": "Build an API", "completed": False},
]
next_id = 3


# GET /tasks — List all tasks
@app.route('/tasks', methods=['GET'])
def get_tasks():
    return jsonify({"tasks": tasks, "count": len(tasks)})


# GET /tasks/<id> — Get a single task
@app.route('/tasks/<int:task_id>', methods=['GET'])
def get_task(task_id):
    task = next((t for t in tasks if t['id'] == task_id), None)
    if task is None:
        return jsonify({"error": "Task not found"}), 404
    return jsonify(task)


# POST /tasks — Create a new task
@app.route('/tasks', methods=['POST'])
def create_task():
    global next_id
    data = request.get_json()
    
    if not data or 'title' not in data:
        return jsonify({"error": "Title is required"}), 400
    
    task = {
        "id": next_id,
        "title": data['title'],
        "completed": data.get('completed', False)
    }
    next_id += 1
    tasks.append(task)
    return jsonify(task), 201


# PUT /tasks/<id> — Update a task
@app.route('/tasks/<int:task_id>', methods=['PUT'])
def update_task(task_id):
    task = next((t for t in tasks if t['id'] == task_id), None)
    if task is None:
        return jsonify({"error": "Task not found"}), 404
    
    data = request.get_json()
    task['title'] = data.get('title', task['title'])
    task['completed'] = data.get('completed', task['completed'])
    return jsonify(task)


# DELETE /tasks/<id> — Delete a task
@app.route('/tasks/<int:task_id>', methods=['DELETE'])
def delete_task(task_id):
    global tasks
    task = next((t for t in tasks if t['id'] == task_id), None)
    if task is None:
        return jsonify({"error": "Task not found"}), 404
    
    tasks = [t for t in tasks if t['id'] != task_id]
    return jsonify({"message": "Task deleted"}), 200


if __name__ == '__main__':
    app.run(debug=True)
```

**Test with curl:**
```bash

curl http://localhost:5000/tasks

# Get task 1
curl http://localhost:5000/tasks/1

# Create a task
curl -X POST http://localhost:5000/tasks \
  -H "Content-Type: application/json" \
  -d '{"title": "Write tests"}'

# Update task 1
curl -X PUT http://localhost:5000/tasks/1 \
  -H "Content-Type: application/json" \
  -d '{"completed": true}'

# Delete task 2
curl -X DELETE http://localhost:5000/tasks/2
```

## Exercise 2: User Authentication Endpoints
**Difficulty:** ★★★☆☆  
**Concepts:** Request hooks (before_request), token-based auth, hashing, error handling  

**Description:**  
Create a simple authentication system with user registration and login endpoints. Implement token-based authentication using Flask's `before_request` hook to protect certain routes.

**Example Starter Code:**
```python
from flask import Flask, request, jsonify, g
import hashlib
import secrets
import time

app = Flask(__name__)

# In-memory storage
users = {}
tokens = {}


def hash_password(password):
    """Simple password hashing (use bcrypt in production!)"""
    return hashlib.sha256(password.encode()).hexdigest()


def generate_token():
    return secrets.token_hex(32)


# Authentication middleware
@app.before_request
def authenticate():
    """Check for valid token on protected routes."""
    # Skip auth for registration and login
    if request.endpoint in ['register', 'login', None]:
        return
    
    token = request.headers.get('Authorization', '').replace('Bearer ', '')
    
    if not token or token not in tokens:
        return jsonify({"error": "Authentication required"}), 401
    
    # Check token expiration (1 hour expiry)
    token_data = tokens[token]
    if time.time() - token_data['created_at'] > 3600:
        del tokens[token]
        return jsonify({"error": "Token expired"}), 401
    
    g.current_user = token_data['username']


# POST /register
@app.route('/register', methods=['POST'])
def register():
    data = request.get_json()
    
    if not data or 'username' not in data or 'password' not in data:
        return jsonify({"error": "Username and password required"}), 400
    
    username = data['username']
    
    if username in users:
        return jsonify({"error": "Username already exists"}), 409
    
    users[username] = {
        "username": username,
        "password_hash": hash_password(data['password']),
        "created_at": time.time()
    }
    
    return jsonify({"message": f"User '{username}' registered successfully"}), 201


# POST /login
@app.route('/login', methods=['POST'])
def login():
    data = request.get_json()
    
    if not data or 'username' not in data or 'password' not in data:
        return jsonify({"error": "Username and password required"}), 400
    
    username = data['username']
    user = users.get(username)
    
    if not user or user['password_hash'] != hash_password(data['password']):
        return jsonify({"error": "Invalid credentials"}), 401
    
    token = generate_token()
    tokens[token] = {
        "username": username,
        "created_at": time.time()
    }
    
    return jsonify({
        "message": "Login successful",
        "token": token,
        "username": username
    })


# GET /profile (protected)
@app.route('/profile', methods=['GET'])
def profile():
    return jsonify({
        "username": g.current_user,
        "message": "This is a protected route"
    })


# GET /users (protected — admin-like)
@app.route('/users', methods=['GET'])
def list_users():
    return jsonify({
        "users": [{"username": u} for u in users],
        "count": len(users)
    })


if __name__ == '__main__':
    app.run(debug=True)
```

**Test with curl:**
```bash

  -H "Content-Type: application/json" \
  -d '{"username": "alice", "password": "secret123"}'

# Login (save the token)
curl -X POST http://localhost:5000/login \
  -H "Content-Type: application/json" \
  -d '{"username": "alice", "password": "secret123"}'

# Access protected route
curl http://localhost:5000/profile \
  -H "Authorization: Bearer <token-from-login>"
```

## Exercise 3: File Upload & Download API
**Difficulty:** ★★★☆☆  
**Concepts:** File handling, content types, error handling, request validation  

**Description:**  
Build an API that supports file uploads and downloads. Validate file types, track file metadata, and serve files back to clients. Include error handling for invalid uploads.

**Example Starter Code:**
```python
from flask import Flask, request, jsonify, send_file
import os
import uuid
import time
from werkzeug.utils import secure_filename

app = Flask(__name__)

# Configuration
UPLOAD_FOLDER = 'uploads'
ALLOWED_EXTENSIONS = {'txt', 'pdf', 'png', 'jpg', 'jpeg', 'gif', 'json', 'csv'}

os.makedirs(UPLOAD_FOLDER, exist_ok=True)
app.config['UPLOAD_FOLDER'] = UPLOAD_FOLDER
app.config['MAX_CONTENT_LENGTH'] = 16 * 1024 * 1024  # 16MB limit

# File metadata storage
file_metadata = {}


def allowed_file(filename):
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS


# Error handler for file too large
@app.errorhandler(413)
def too_large(e):
    return jsonify({"error": "File is too large (max 16MB)"}), 413


# GET /files — List all files
@app.route('/files', methods=['GET'])
def list_files():
    files = []
    for file_id, meta in file_metadata.items():
        files.append({
            "id": file_id,
            "original_name": meta['original_name'],
            "size": meta['size'],
            "type": meta['mime_type'],
            "uploaded_at": meta['uploaded_at']
        })
    return jsonify({"files": files, "count": len(files)})


# POST /upload — Upload a file
@app.route('/upload', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return jsonify({"error": "No file part in request"}), 400
    
    file = request.files['file']
    
    if file.filename == '':
        return jsonify({"error": "No file selected"}), 400
    
    if not allowed_file(file.filename):
        return jsonify({
            "error": f"File type not allowed. Allowed: {', '.join(ALLOWED_EXTENSIONS)}"
        }), 400
    
    # Secure the filename and generate unique ID
    original_name = secure_filename(file.filename)
    extension = original_name.rsplit('.', 1)[1].lower()
    file_id = str(uuid.uuid4())[:8]
    saved_name = f"{file_id}.{extension}"
    
    filepath = os.path.join(app.config['UPLOAD_FOLDER'], saved_name)
    file.save(filepath)
    
    file_size = os.path.getsize(filepath)
    
    # Store metadata
    file_metadata[file_id] = {
        "original_name": original_name,
        "saved_name": saved_name,
        "path": filepath,
        "size": file_size,
        "mime_type": file.content_type,
        "uploaded_at": time.time()
    }
    
    return jsonify({
        "message": "File uploaded successfully",
        "file_id": file_id,
        "original_name": original_name,
        "size": file_size
    }), 201


# GET /download/<file_id> — Download a file
@app.route('/download/<file_id>', methods=['GET'])
def download_file(file_id):
    if file_id not in file_metadata:
        return jsonify({"error": "File not found"}), 404
    
    meta = file_metadata[file_id]
    return send_file(
        meta['path'],
        mimetype=meta['mime_type'],
        as_attachment=True,
        download_name=meta['original_name']
    )


# GET /files/<file_id> — Get file info
@app.route('/files/<file_id>', methods=['GET'])
def file_info(file_id):
    if file_id not in file_metadata:
        return jsonify({"error": "File not found"}), 404
    
    meta = file_metadata[file_id]
    return jsonify({
        "id": file_id,
        "original_name": meta['original_name'],
        "size": meta['size'],
        "type": meta['mime_type'],
        "uploaded_at": meta['uploaded_at']
    })


# DELETE /files/<file_id> — Delete a file
@app.route('/files/<file_id>', methods=['DELETE'])
def delete_file(file_id):
    if file_id not in file_metadata:
        return jsonify({"error": "File not found"}), 404
    
    meta = file_metadata[file_id]
    
    try:
        os.remove(meta['path'])
    except OSError:
        pass
    
    del file_metadata[file_id]
    return jsonify({"message": "File deleted successfully"})


if __name__ == '__main__':
    app.run(debug=True)
```
