## Exercise 1: Basic GET and POST Endpoints with Pydantic

**Title:** Your First FastAPI — Path Operations and Pydantic Models

**Description:** Create a FastAPI application with a root GET endpoint and a POST endpoint that accepts and validates a Pydantic model. Learn path operations, type hints, and automatic API documentation.

**Difficulty:** ★★☆☆☆ (Easy)

**Concepts Practiced:**
- `FastAPI()` app creation
- Path operation decorators (`@app.get`, `@app.post`)
- Pydantic `BaseModel` for request bodies
- Type hints and automatic validation
- Automatic interactive docs (`/docs`)
- `uvicorn` server

**Starter Code:**
```python
# Exercise 1: Your First FastAPI
from fastapi import FastAPI
from pydantic import BaseModel
from typing import Optional

app = FastAPI(title="My First API", version="1.0.0")

class Item(BaseModel):
    name: str
    price: float
    description: Optional[str] = None
    tax: Optional[float] = None

@app.get("/")
def root():
    return {"message": "Hello, FastAPI!", "docs": "/docs"}

@app.get("/hello/{name}")
def greet(name: str, age: Optional[int] = None):
    response = {"message": f"Hello, {name}!"}
    if age:
        response["age"] = age
        response["birth_year"] = 2026 - age
    return response

@app.post("/items/")
def create_item(item: Item):
    price_with_tax = item.price
    if item.tax:
        price_with_tax += item.tax

    return {
        "item": item.model_dump(),
        "price_with_tax": price_with_tax,
        "message": f"Created item: {item.name}"
    }
```

## Exercise 2: CRUD API for a Book Store

**Title:** Book Store — Full CRUD with Path and Query Parameters

**Description:** Build a full CRUD API for managing a book store. Implement endpoints to create, read, update, and delete books. Use path parameters for resource IDs, query parameters for filtering, and proper HTTP status codes. Store data in-memory.

**Difficulty:** ★★★☆☆ (Moderate)

**Concepts Practiced:**
- Full CRUD operations
- Path parameters (`{book_id}`)
- Query parameters for filtering
- `HTTPException` for error responses
- Pydantic models for request/response
- `List` response model
- Enum types for genres

**Starter Code:**
```python
# Exercise 2: Book Store CRUD API
from fastapi import FastAPI, HTTPException, Query
from pydantic import BaseModel
from enum import Enum
from typing import List, Optional

app = FastAPI(title="Book Store API")

class Genre(str, Enum):
    FICTION = "fiction"
    NON_FICTION = "non-fiction"
    SCIFI = "sci-fi"
    FANTASY = "fantasy"
    MYSTERY = "mystery"

class Book(BaseModel):
    id: int
    title: str
    author: str
    genre: Genre
    year: int
    available: bool = True

class BookCreate(BaseModel):
    title: str
    author: str
    genre: Genre
    year: int

class BookUpdate(BaseModel):
    title: Optional[str] = None
    author: Optional[str] = None
    genre: Optional[Genre] = None
    year: Optional[int] = None
    available: Optional[bool] = None

# In-memory database
books: List[Book] = [
    Book(id=1, title="Dune", author="Frank Herbert", genre=Genre.SCIFI, year=1965),
    Book(id=2, title="1984", author="George Orwell", genre=Genre.FICTION, year=1949),
]
next_id = 3

@app.get("/books", response_model=List[Book])
def get_books(
    genre: Optional[Genre] = Query(None, description="Filter by genre"),
    available: Optional[bool] = Query(None, description="Filter by availability"),
    author: Optional[str] = Query(None, description="Search by author name")
):
    result = books
    if genre:
        result = [b for b in result if b.genre == genre]
    if available is not None:
        result = [b for b in result if b.available == available]
    if author:
        result = [b for b in result if author.lower() in b.author.lower()]
    return result

@app.get("/books/{book_id}", response_model=Book)
def get_book(book_id: int):
    book = next((b for b in books if b.id == book_id), None)
    if not book:
        raise HTTPException(status_code=404, detail=f"Book {book_id} not found")
    return book

@app.post("/books", response_model=Book, status_code=201)
def create_book(book_data: BookCreate):
    global next_id
    book = Book(id=next_id, **book_data.model_dump())
    next_id += 1
    books.append(book)
    return book

@app.put("/books/{book_id}", response_model=Book)
def update_book(book_id: int, book_data: BookUpdate):
    book = next((b for b in books if b.id == book_id), None)
    if not book:
        raise HTTPException(status_code=404, detail=f"Book {book_id} not found")

    update_data = book_data.model_dump(exclude_unset=True)
    for key, value in update_data.items():
        setattr(book, key, value)
    return book

@app.delete("/books/{book_id}")
def delete_book(book_id: int):
    global books
    book = next((b for b in books if b.id == book_id), None)
    if not book:
        raise HTTPException(status_code=404, detail=f"Book {book_id} not found")

    books = [b for b in books if b.id != book_id]
    return {"message": f"Book '{book.title}' deleted", "remaining": len(books)}
```

## Exercise 3: Query Parameters, Validation, and Response Models

**Title:** Advanced Querying — Pagination, Sorting, and Validation

**Description:** Extend the Book Store API with advanced query features: pagination, sorting, field selection, and request validation. Use FastAPI's `Query`, `Path`, and response models to create a production-quality search endpoint.

**Difficulty:** ★★★☆☆ (Moderate)

**Concepts Practiced:**
- Query parameter validation (`Query(ge=..., le=...)`)
- Path parameter validation (`Path(...)`)
- Pagination with `skip` and `limit`
- Response model filtering
- Multiple query parameters
- Custom response models

**Starter Code:**
```python
# Exercise 3: Advanced Querying
from fastapi import FastAPI, HTTPException, Query, Path
from pydantic import BaseModel, Field
from typing import List, Optional

app = FastAPI(title="Advanced Book API")

class BookInDB(BaseModel):
    id: int
    title: str
    author: str
    genre: str
    year: int
    price: float
    rating: float = Field(ge=0, le=5)
    available: bool

# Sample data
books_db = [
    BookInDB(id=1, title="Dune", author="Frank Herbert", genre="sci-fi", year=1965, price=12.99, rating=4.8, available=True),
    BookInDB(id=2, title="1984", author="George Orwell", genre="fiction", year=1949, price=9.99, rating=4.7, available=True),
    BookInDB(id=3, title="The Hobbit", author="J.R.R. Tolkien", genre="fantasy", year=1937, price=14.99, rating=4.9, available=False),
    BookInDB(id=4, title="Foundation", author="Isaac Asimov", genre="sci-fi", year=1951, price=11.99, rating=4.6, available=True),
    BookInDB(id=5, title="Pride and Prejudice", author="Jane Austen", genre="fiction", year=1813, price=7.99, rating=4.5, available=True),
]

class BookSummary(BaseModel):
    id: int
    title: str
    author: str
    rating: float

class PaginatedResponse(BaseModel):
    total: int
    page: int
    page_size: int
    total_pages: int
    books: List[BookSummary]

@app.get("/books/search", response_model=PaginatedResponse)
def search_books(
    q: Optional[str] = Query(None, description="Search in title/author"),
    genre: Optional[str] = Query(None, description="Filter by genre"),
    min_rating: float = Query(0, ge=0, le=5, description="Minimum rating"),
    min_year: Optional[int] = Query(None, description="Published after year"),
    sort_by: str = Query("title", regex="^(title|author|year|rating|price)$"),
    sort_order: str = Query("asc", regex="^(asc|desc)$"),
    page: int = Query(1, ge=1, description="Page number"),
    page_size: int = Query(10, ge=1, le=50, description="Items per page"),
):
    # Apply filters
    result = [b for b in books_db]

    if q:
        q_lower = q.lower()
        result = [b for b in result if q_lower in b.title.lower() or q_lower in b.author.lower()]
    if genre:
        result = [b for b in result if b.genre == genre.lower()]
    if min_rating > 0:
        result = [b for b in result if b.rating >= min_rating]
    if min_year:
        result = [b for b in result if b.year >= min_year]

    # Sort
    reverse = (sort_order == "desc")
    result.sort(key=lambda b: getattr(b, sort_by), reverse=reverse)

    # Paginate
    total = len(result)
    total_pages = max(1, (total + page_size - 1) // page_size)
    start = (page - 1) * page_size
    end = start + page_size
    page_items = result[start:end]

    # Convert to summary
    summaries = [BookSummary(id=b.id, title=b.title, author=b.author, rating=b.rating) for b in page_items]

    return PaginatedResponse(
        total=total,
        page=page,
        page_size=page_size,
        total_pages=total_pages,
        books=summaries
    )

@app.get("/books/{book_id}", response_model=BookInDB)
def get_book_detail(
    book_id: int = Path(..., ge=1, description="The ID of the book")
):
    book = next((b for b in books_db if b.id == book_id), None)
    if not book:
        raise HTTPException(status_code=404, detail=f"Book {book_id} not found")
    return book

@app.get("/stats")
def get_stats():
    genres = {}
    for b in books_db:
        genres[b.genre] = genres.get(b.genre, 0) + 1

    return {
        "total_books": len(books_db),
        "average_rating": round(sum(b.rating for b in books_db) / len(books_db), 2),
        "average_price": round(sum(b.price for b in books_db) / len(books_db), 2),
        "genres": genres,
        "available_count": sum(1 for b in books_db if b.available),
    }
```
