## Exercise 1: CRUD API with Pydantic Models
**Difficulty:** ★★★☆☆  
**Concepts:** FastAPI app, Pydantic models, path/query params, auto-docs, type hints  

**Description:**  
Build a CRUD API for managing a collection of books using FastAPI's type-driven approach. Use Pydantic models for request/response validation. FastAPI auto-generates OpenAPI docs at `/docs`.

**Example Starter Code:**
```python
from fastapi import FastAPI, HTTPException, status
from pydantic import BaseModel, Field
from typing import Optional

app = FastAPI(title="Book API", version="1.0.0")


# Pydantic models
class Book(BaseModel):
    id: Optional[int] = None
    title: str = Field(..., min_length=1, max_length=200)
    author: str = Field(..., min_length=1, max_length=100)
    pages: int = Field(..., gt=0, le=5000)
    isbn: str = Field(..., min_length=10, max_length=13)


class BookUpdate(BaseModel):
    title: Optional[str] = Field(None, min_length=1, max_length=200)
    author: Optional[str] = Field(None, min_length=1, max_length=100)
    pages: Optional[int] = Field(None, gt=0, le=5000)


# In-memory storage
books_db = {
    1: {"id": 1, "title": "1984", "author": "George Orwell", "pages": 328, "isbn": "9780451524935"},
    2: {"id": 2, "title": "Dune", "author": "Frank Herbert", "pages": 412, "isbn": "9780441172719"},
}
next_id = 3


@app.get("/")
async def root():
    return {"message": "Welcome to the Book API", "docs": "/docs"}


# GET /books — List all books with optional filtering
@app.get("/books")
async def list_books(author: Optional[str] = None, min_pages: Optional[int] = None):
    books = list(books_db.values())
    
    if author:
        books = [b for b in books if author.lower() in b['author'].lower()]
    if min_pages:
        books = [b for b in books if b['pages'] >= min_pages]
    
    return {"books": books, "count": len(books)}


# GET /books/{book_id} — Get a specific book
@app.get("/books/{book_id}")
async def get_book(book_id: int):
    if book_id not in books_db:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Book with id {book_id} not found"
        )
    return books_db[book_id]


# POST /books — Create a new book
@app.post("/books", status_code=status.HTTP_201_CREATED)
async def create_book(book: Book):
    global next_id
    
    # Check for duplicate ISBN
    for existing in books_db.values():
        if existing['isbn'] == book.isbn:
            raise HTTPException(
                status_code=status.HTTP_409_CONFLICT,
                detail=f"Book with ISBN {book.isbn} already exists"
            )
    
    book.id = next_id
    books_db[next_id] = book.model_dump()
    next_id += 1
    
    return books_db[book.id]


# PUT /books/{book_id} — Update a book
@app.put("/books/{book_id}")
async def update_book(book_id: int, book_update: BookUpdate):
    if book_id not in books_db:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Book with id {book_id} not found"
        )
    
    # Update only provided fields
    update_data = book_update.model_dump(exclude_unset=True)
    books_db[book_id].update(update_data)
    
    return books_db[book_id]


# DELETE /books/{book_id} — Delete a book
@app.delete("/books/{book_id}")
async def delete_book(book_id: int):
    if book_id not in books_db:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=f"Book with id {book_id} not found"
        )
    
    deleted = books_db.pop(book_id)
    return {"message": f"Book '{deleted['title']}' deleted"}
```

## Exercise 2: Async Endpoints with Database Simulation
**Difficulty:** ★★★☆☆  
**Concepts:** Async/await, dependency injection, background tasks, middleware  

**Description:**  
Create async endpoints that simulate database operations. Use FastAPI's dependency injection for a mock database session. Add a background task for sending notifications and a custom middleware for request logging.

**Example Starter Code:**
```python
from fastapi import FastAPI, Depends, BackgroundTasks, Request
from pydantic import BaseModel, EmailStr
from typing import Optional
import asyncio
import time

app = FastAPI(title="Async User API")


# --- Middleware ---
@app.middleware("http")
async def log_requests(request: Request, call_next):
    """Log all incoming requests with timing."""
    start_time = time.time()
    response = await call_next(request)
    duration = time.time() - start_time
    
    print(f"{request.method} {request.url.path} — {response.status_code} — {duration:.3f}s")
    response.headers["X-Process-Time"] = str(duration)
    return response


# --- Pydantic Models ---
class UserCreate(BaseModel):
    name: str
    email: EmailStr
    age: Optional[int] = None


class UserResponse(BaseModel):
    id: int
    name: str
    email: str
    age: Optional[int] = None


# --- Mock Database (Dependency) ---
class MockDB:
    def __init__(self):
        self.users = {}
        self.next_id = 1
    
    async def add_user(self, user: UserCreate) -> dict:
        # Simulate async database operation
        await asyncio.sleep(0.1)
        user_data = user.model_dump()
        user_data['id'] = self.next_id
        self.users[self.next_id] = user_data
        self.next_id += 1
        return user_data
    
    async def get_user(self, user_id: int) -> Optional[dict]:
        await asyncio.sleep(0.05)
        return self.users.get(user_id)
    
    async def get_all_users(self) -> list:
        await asyncio.sleep(0.05)
        return list(self.users.values())


db = MockDB()


async def get_db():
    """Dependency injection: provide database access."""
    return db


# --- Background Task ---
async def send_welcome_email(email: str, name: str):
    """Simulate sending a welcome email (runs in background)."""
    await asyncio.sleep(2)  # Simulate email sending
    print(f"📧 Welcome email sent to {name} at {email}")


# --- Routes ---
@app.get("/")
async def root():
    return {"message": "Async User API", "docs": "/docs"}


@app.get("/users", response_model=list[UserResponse])
async def list_users(db: MockDB = Depends(get_db)):
    users = await db.get_all_users()
    return users


@app.get("/users/{user_id}", response_model=UserResponse)
async def get_user(user_id: int, db: MockDB = Depends(get_db)):
    user = await db.get_user(user_id)
    if not user:
        from fastapi import HTTPException
        raise HTTPException(status_code=404, detail="User not found")
    return user


@app.post("/users", response_model=UserResponse, status_code=201)
async def create_user(
    user: UserCreate,
    background_tasks: BackgroundTasks,
    db: MockDB = Depends(get_db)
):
    new_user = await db.add_user(user)
    
    # Schedule background task
    background_tasks.add_task(send_welcome_email, user.email, user.name)
    
    return new_user


@app.delete("/users/{user_id}")
async def delete_user(user_id: int, db: MockDB = Depends(get_db)):
    user = await db.get_user(user_id)
    if not user:
        from fastapi import HTTPException
        raise HTTPException(status_code=404, detail="User not found")
    
    del db.users[user_id]
    return {"message": f"User '{user['name']}' deleted"}
```

## Exercise 3: Input Validation & Error Handling
**Difficulty:** ★★★☆☆  
**Concepts:** Custom validators, field constraints, global exception handlers, response models  

**Description:**  
Create a product catalog API with advanced Pydantic validation (custom validators, regex, enums), global exception handlers, and proper HTTP status codes. Include query parameter validation.

**Example Starter Code:**
```python
from fastapi import FastAPI, HTTPException, Query, Request
from fastapi.responses import JSONResponse
from pydantic import BaseModel, Field, field_validator
from enum import Enum
from typing import Optional
from datetime import datetime
import re

app = FastAPI(title="Product Catalog API")


# --- Enums ---
class Category(str, Enum):
    ELECTRONICS = "electronics"
    CLOTHING = "clothing"
    BOOKS = "books"
    FOOD = "food"


# --- Pydantic Models with Validators ---
class ProductCreate(BaseModel):
    name: str = Field(..., min_length=2, max_length=100, examples=["Wireless Mouse"])
    category: Category
    price: float = Field(..., gt=0, le=100000)
    sku: str = Field(..., min_length=6, max_length=20, pattern=r'^[A-Z]{3}-\d{3}$',
                     examples=["ABC-123"])
    description: Optional[str] = Field(None, max_length=500)
    
    @field_validator('sku')
    @classmethod
    def validate_sku(cls, v: str) -> str:
        """Custom validator for SKU format."""
        pattern = r'^[A-Z]{3}-\d{3}$'
        if not re.match(pattern, v):
            raise ValueError('SKU must match format: XXX-000 (3 uppercase letters, hyphen, 3 digits)')
        return v

    @field_validator('name')
    @classmethod
    def name_must_not_be_empty(cls, v: str) -> str:
        if not v.strip():
            raise ValueError('Name cannot be empty or whitespace')
        return v.strip()


class ProductResponse(BaseModel):
    id: int
    name: str
    category: Category
    price: float
    sku: str
    description: Optional[str] = None
    created_at: str


# --- Global Exception Handlers ---
@app.exception_handler(ValueError)
async def value_error_handler(request: Request, exc: ValueError):
    return JSONResponse(
        status_code=422,
        content={"error": "Validation Error", "detail": str(exc)}
    )


@app.exception_handler(404)
async def not_found_handler(request: Request, exc: HTTPException):
    return JSONResponse(
        status_code=404,
        content={"error": "Not Found", "detail": str(exc.detail)}
    )


# --- Storage ---
products_db = {}
next_id = 1


# --- Routes ---
@app.get("/", response_model=dict)
async def root():
    return {"message": "Product Catalog API", "endpoints": ["/products", "/products/{id}"]}


@app.get("/products", response_model=list[ProductResponse])
async def list_products(
    category: Optional[Category] = Query(None, description="Filter by category"),
    min_price: Optional[float] = Query(None, ge=0, description="Minimum price"),
    max_price: Optional[float] = Query(None, ge=0, description="Maximum price"),
    search: Optional[str] = Query(None, min_length=2, description="Search in name")
):
    products = list(products_db.values())
    
    if category:
        products = [p for p in products if p['category'] == category.value]
    if min_price is not None:
        products = [p for p in products if p['price'] >= min_price]
    if max_price is not None:
        products = [p for p in products if p['price'] <= max_price]
    if search:
        products = [p for p in products if search.lower() in p['name'].lower()]
    
    return products


@app.get("/products/{product_id}", response_model=ProductResponse)
async def get_product(product_id: int):
    if product_id not in products_db:
        raise HTTPException(status_code=404, detail=f"Product {product_id} not found")
    return products_db[product_id]


@app.post("/products", response_model=ProductResponse, status_code=201)
async def create_product(product: ProductCreate):
    global next_id
    
    # Check for duplicate SKU
    for existing in products_db.values():
        if existing['sku'] == product.sku:
            raise HTTPException(
                status_code=409,
                detail=f"Product with SKU '{product.sku}' already exists"
            )
    
    product_data = product.model_dump()
    product_data['id'] = next_id
    product_data['created_at'] = datetime.now().isoformat()
    products_db[next_id] = product_data
    next_id += 1
    
    return product_data


@app.get("/health")
async def health_check():
    return {
        "status": "healthy",
        "products_count": len(products_db),
        "timestamp": datetime.now().isoformat()
    }
```
