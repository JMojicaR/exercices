## Exercise 1: Basic CRUD with Pydantic Models
- **Difficulty**: Basic
- **Concepts**: FastAPI app, Pydantic BaseModel, path/query parameters, automatic docs (Swagger), type hints, HTTP status codes
- **Description**: Build a CRUD API for a book collection. Use Pydantic models for request and response schemas. Each book has an `id`, `title`, `author`, `year`, and `is_available` status. Support GET (all/single), POST, PUT, and DELETE. FastAPI auto-generates interactive docs at `/docs`.

### Example/Starter Code
```python
from fastapi import FastAPI, HTTPException, status
from pydantic import BaseModel, Field
from typing import Optional, List

app = FastAPI(title="Book API", version="1.0")

# --- Pydantic Models ---
class BookCreate(BaseModel):
    title: str = Field(..., min_length=1, max_length=200, description="Book title")
    author: str = Field(..., min_length=1, max_length=100)
    year: int = Field(..., ge=1000, le=2100)

class BookUpdate(BaseModel):
    title: Optional[str] = Field(None, min_length=1, max_length=200)
    author: Optional[str] = Field(None, min_length=1, max_length=100)
    year: Optional[int] = Field(None, ge=1000, le=2100)
    is_available: Optional[bool] = None

class Book(BookCreate):
    id: int
    is_available: bool = True

# --- In-memory storage ---
books: List[Book] = []
next_id = 1

@app.get("/books", response_model=List[Book])
async def get_books(author: Optional[str] = None):
    """Get all books, optionally filtered by author."""
    # TODO: If author query param is provided, filter by author (case-insensitive)
    if author:
        return [b for b in books if author.lower() in b.author.lower()]
    return books

@app.get("/books/{book_id}", response_model=Book)
async def get_book(book_id: int):
    # TODO: Find book by ID, raise HTTPException(404) if not found
    for book in books:
        if book.id == book_id:
            return book
    raise HTTPException(status_code=404, detail="Book not found")

@app.post("/books", response_model=Book, status_code=status.HTTP_201_CREATED)
async def create_book(book: BookCreate):
    global next_id
    new_book = Book(id=next_id, **book.model_dump())
    books.append(new_book)
    next_id += 1
    return new_book

@app.put("/books/{book_id}", response_model=Book)
async def update_book(book_id: int, update: BookUpdate):
    # TODO: Find book, update only non-None fields, raise 404 if not found
    for i, book in enumerate(books):
        if book.id == book_id:
            update_data = update.model_dump(exclude_unset=True)
            updated = book.model_copy(update=update_data)
            books[i] = updated
            return updated
    raise HTTPException(status_code=404, detail="Book not found")

@app.delete("/books/{book_id}", status_code=status.HTTP_204_NO_CONTENT)
async def delete_book(book_id: int):
    # TODO: Remove book, raise 404 if not found
    for i, book in enumerate(books):
        if book.id == book_id:
            books.pop(i)
            return
    raise HTTPException(status_code=404, detail="Book not found")
```

## Exercise 2: Path Parameters, Query Parameters, and Advanced Validation
- **Difficulty**: Basic
- **Concepts**: Path validators (ge/le/gt/lt), query parameter validation, enum types, response filtering, custom validators
- **Description**: Create an API for querying a product catalog with advanced filtering. Use path parameters with constraints, query parameters with defaults and ranges, and enum-based filtering. Implement a `@field_validator` for custom business logic on the Pydantic model.

### Example/Starter Code
```python
from fastapi import FastAPI, HTTPException, Query, Path
from pydantic import BaseModel, Field, field_validator
from enum import Enum
from typing import Optional, List

app = FastAPI()

class Category(str, Enum):
    ELECTRONICS = "electronics"
    BOOKS = "books"
    CLOTHING = "clothing"
    FOOD = "food"

class Product(BaseModel):
    id: int
    name: str = Field(..., min_length=2, max_length=100)
    price: float = Field(..., gt=0)
    category: Category
    in_stock: bool = True

    @field_validator('name')
    @classmethod
    def name_must_not_be_empty(cls, v: str) -> str:
        if not v.strip():
            raise ValueError('Name cannot be empty or whitespace-only')
        return v.strip()

class ProductCreate(BaseModel):
    name: str = Field(..., min_length=2, max_length=100)
    price: float = Field(..., gt=0, description="Must be greater than 0")
    category: Category
    in_stock: bool = True

    @field_validator('name')
    @classmethod
    def name_must_not_be_empty(cls, v: str) -> str:
        if not v.strip():
            raise ValueError('Name cannot be empty or whitespace-only')
        return v.strip()

products = [
    Product(id=1, name="Laptop", price=999.99, category=Category.ELECTRONICS),
    Product(id=2, name="T-Shirt", price=19.99, category=Category.CLOTHING),
    Product(id=3, name="Python Book", price=39.99, category=Category.BOOKS),
]

@app.get("/products")
async def search_products(
    category: Optional[Category] = None,
    min_price: Optional[float] = Query(None, ge=0, description="Minimum price filter"),
    max_price: Optional[float] = Query(None, ge=0, description="Maximum price filter"),
    in_stock: Optional[bool] = None,
    search: Optional[str] = Query(None, min_length=1, description="Search in name"),
):
    """Search products with multiple optional filters."""
    # TODO: Implement multi-filter logic
    # - Filter by category (if provided)
    # - Filter by price range (if min_price or max_price provided)
    # - Filter by stock status (if in_stock provided)
    # - Search by name (case-insensitive substring match)
    result = products
    if category:
        result = [p for p in result if p.category == category]
    if min_price is not None:
        result = [p for p in result if p.price >= min_price]
    if max_price is not None:
        result = [p for p in result if p.price <= max_price]
    if in_stock is not None:
        result = [p for p in result if p.in_stock == in_stock]
    if search:
        result = [p for p in result if search.lower() in p.name.lower()]
    return result

@app.get("/products/{product_id}")
async def get_product(
    product_id: int = Path(..., ge=1, description="Product ID must be >= 1")
):
    # TODO: Return product or 404
    for p in products:
        if p.id == product_id:
            return p
    raise HTTPException(status_code=404, detail="Product not found")
```

## Exercise 3: Dependency Injection and Global Error Handling
- **Difficulty**: Basic
- **Concepts**: Depends() for dependency injection, middleware, exception handlers, background tasks, response headers
- **Description**: Build a user API that uses dependency injection for a simulated database session and authentication. Add a global exception handler for validation errors. Use background tasks for sending a "welcome email" after user creation. Implement rate-limiting via a simple dependency.

### Example/Starter Code
```python
from fastapi import FastAPI, Depends, HTTPException, Request, BackgroundTasks
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from pydantic import BaseModel, EmailStr
from typing import Optional, Dict
import time

app = FastAPI()

# --- Fake Database ---
class FakeDB:
    def __init__(self):
        self.users: Dict[int, dict] = {}
        self._next_id = 1

    def add_user(self, user_data: dict) -> dict:
        user_data["id"] = self._next_id
        self.users[self._next_id] = user_data
        self._next_id += 1
        return user_data

    def get_user(self, user_id: int) -> Optional[dict]:
        return self.users.get(user_id)

    def get_all_users(self) -> list:
        return list(self.users.values())

db = FakeDB()

# --- Dependency: Get DB ---
def get_db():
    """Dependency that provides a database session."""
    # In a real app this would be `yield` with a session
    return db

# --- Dependency: Simple Rate Limiter ---
request_counts: Dict[str, list] = {}

def rate_limiter(request: Request, max_requests: int = 5, window: float = 60.0):
    """Allow max_requests per window seconds per client IP."""
    client_ip = request.client.host
    now = time.time()

    if client_ip not in request_counts:
        request_counts[client_ip] = []

    # Clean old timestamps
    request_counts[client_ip] = [
        t for t in request_counts[client_ip] if now - t < window
    ]

    if len(request_counts[client_ip]) >= max_requests:
        raise HTTPException(status_code=429, detail="Too many requests")

    request_counts[client_ip].append(now)

# --- Global Exception Handler ---
@app.exception_handler(RequestValidationError)
async def validation_exception_handler(request: Request, exc: RequestValidationError):
    return JSONResponse(
        status_code=422,
        content={
            "error": "Validation failed",
            "details": exc.errors(),
        }
    )

# --- Models ---
class UserCreate(BaseModel):
    email: EmailStr
    name: str

class UserResponse(BaseModel):
    id: int
    email: str
    name: str

# --- Background Task ---
def send_welcome_email(email: str, name: str):
    """Simulate sending a welcome email."""
    print(f"[BG TASK] Sending welcome email to {name} <{email}>...")
    time.sleep(1)  # Simulate network delay
    print(f"[BG TASK] Welcome email sent to {email}!")

# --- Routes ---
@app.get("/users", response_model=list[UserResponse])
async def get_users(db: FakeDB = Depends(get_db)):
    return db.get_all_users()

@app.post("/users", response_model=UserResponse, status_code=201)
async def create_user(
    user: UserCreate,
    background_tasks: BackgroundTasks,
    db: FakeDB = Depends(get_db),
):
    # TODO: Check if email already exists, raise 409 Conflict if so
    created = db.add_user(user.model_dump())
    background_tasks.add_task(send_welcome_email, created["email"], created["name"])
    return created

@app.get("/users/{user_id}", response_model=UserResponse)
async def get_user(
    user_id: int,
    db: FakeDB = Depends(get_db),
    _rate: None = Depends(rate_limiter),  # Apply rate limiting
):
    # TODO: Look up user, raise 404 if not found
    user = db.get_user(user_id)
    if user is None:
        raise HTTPException(status_code=404, detail="User not found")
    return user
```
