## Exercise 2: Middleware — Request Logger and Error Handler

**Title:** Build Custom Logging and Error Handling Middleware

**Description:**  
Create two custom Express middleware functions:
1. **`requestLogger`** — logs each request with: timestamp, HTTP method, URL, response status code, and response time (in ms). Use `res.on('finish', ...)` to capture the status code after the response is sent.
2. **`errorHandler`** — an error-handling middleware (4 parameters) that catches errors, logs them, and returns a JSON error response `{ error: message, status: code }`.

Also create a middleware `validateProduct` that validates product data in `POST` and `PUT` requests, returning 400 with a descriptive error if validation fails.

Apply these to a simple Products API.

**Concepts Practiced:** Custom middleware, error-handling middleware (4 params), `next()`, `res.on('finish')`, request validation

**Difficulty:** Basic

**Starter Code:**
```javascript
const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

// --- Custom Middleware ---

// 1. Request Logger
function requestLogger(req, res, next) {
    const start = Date.now();

    // When response finishes, log the details
    res.on('finish', () => {
        const duration = Date.now() - start;
        // Log: [timestamp] METHOD /url - STATUS (Xms)
        console.log(`${new Date().toISOString()} ${req.method} ${req.originalUrl} - ${res.statusCode} (${duration}ms)`);
    });

    next();
}

// 2. Product Validation Middleware
function validateProduct(req, res, next) {
    // For POST and PUT requests
    if (['POST', 'PUT'].includes(req.method)) {
        const { name, price } = req.body;

        const errors = [];
        if (!name || typeof name !== 'string' || name.trim() === '') {
            errors.push('"name" is required and must be a non-empty string');
        }
        if (price === undefined || typeof price !== 'number' || price < 0) {
            errors.push('"price" is required and must be a non-negative number');
        }

        if (errors.length > 0) {
            return res.status(400).json({ error: 'Validation failed', details: errors });
        }
    }
    next();
}

// 3. Error Handling Middleware (4 params)
function errorHandler(err, req, res, next) {
    // Log the error
    console.error(`[ERROR] ${err.message}`);

    // Determine status code (use err.status if set, otherwise 500)
    const status = err.status || 500;

    // Return JSON error response
    res.status(status).json({
        error: err.message || 'Internal Server Error',
        status: status
    });
}

// --- Apply Middleware ---
app.use(requestLogger);
app.use('/api/products', validateProduct);  // Apply validation to product routes only

// --- Product Routes (same as Exercise 1, simplified) ---
const products = [];
let nextId = 1;

app.get('/api/products', (req, res) => {
    res.json(products);
});

app.post('/api/products', (req, res) => {
    const product = { id: nextId++, ...req.body, inStock: req.body.inStock ?? true };
    products.push(product);
    res.status(201).json(product);
});

// Test error route
app.get('/api/error', (req, res, next) => {
    const err = new Error('Something went wrong!');
    err.status = 500;
    next(err);  // Pass to error handler
});

// --- Apply Error Handler (must be last) ---
app.use(errorHandler);

app.listen(PORT, () => {
    console.log(`Server running on http://localhost:${PORT}`);
});
```
