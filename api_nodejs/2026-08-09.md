## Exercise 1: Basic Server & GET Endpoint — Your First Express API

### Problem

Set up a basic Express.js server and create your first API endpoint. The app should:

- Listen on port 5000
- Use `express.json()` middleware to parse incoming JSON request bodies
- `GET /` — returns the string `"HELLO FROM HOMEPAGE"`
- `GET /users` — returns a mock list of users as JSON from an in-memory array

Create a `routes/users.js` file that uses `express.Router()` to define the `/users` route, then mount it in `index.js` with `app.use('/users', userRoutes)`.

### Concepts

- `express()` application initialization
- `app.listen(PORT, callback)` to start the server
- `express.json()` middleware for JSON body parsing
- `express.Router()` for modular route definitions
- `app.use()` to mount routers at a path prefix
- `res.send()` to send responses
- ES module syntax (`import`/`export`) with `"type": "module"` in `package.json`

### HTTP Method(s)

`GET`

### Example/Starter Code

**package.json:**
```json
{
  "name": "express-api-exercises",
  "version": "1.0.0",
  "type": "module",
  "scripts": {
    "start": "node index.js"
  },
  "dependencies": {
    "express": "^4.18.0"
  }
}
```

**index.js:**
```javascript
import express from 'express';
import userRoutes from './routes/users.js';

const app = express();
const PORT = 5000;

app.use(express.json());

// Mount user routes at /users
app.use('/users', userRoutes);

// Root endpoint
app.get('/', (req, res) => {
    console.log('[GET ROUTE]');
    res.send('HELLO FROM HOMEPAGE');
});

app.listen(PORT, () =>
    console.log(`Server running on port: http://localhost:${PORT}`)
);
```

**routes/users.js:**
```javascript
import express from 'express';

const router = express.Router();

// Mock database
const users = [
    {
        first_name: 'John',
        last_name: 'Doe',
        email: 'johndoe@example.com',
    },
    {
        first_name: 'Alice',
        last_name: 'Smith',
        email: 'alicesmith@example.com',
    },
];

// GET all users
router.get('/', (req, res) => {
    res.send(users);
});

export default router;
```

**Setup and run:**
```bash
npm init -y
# Add "type": "module" to package.json manually
npm install express
node index.js
# Visit http://localhost:5000/users in browser or Postman
```

## Exercise 2: Full CRUD with In-Memory Storage & UUIDs

### Problem

Extend the users API from Exercise 1 to support full CRUD operations using `uuid` for unique IDs. All data is stored in an in-memory array (lost on server restart). Implement:

- `GET /users` — returns all users
- `POST /users` — creates a new user from `{first_name, last_name, email}` in the request body, assigns a UUID as `id`, appends to the array, and returns a confirmation message
- `GET /users/:id` — finds and returns a single user by UUID, or returns an error if not found
- `DELETE /users/:id` — removes a user by UUID using `Array.filter()`, returns confirmation

Import `v4 as uuidv4` from the `uuid` package for ID generation.

### Concepts

- `req.body` to access JSON data from POST/PATCH requests
- `req.params` to extract dynamic route parameters (`:id`)
- `uuid` package for generating unique identifiers
- Array methods: `push()`, `find()`, `filter()` for CRUD operations
- Spread operator `...user` for copying objects
- Template literals for response messages

### HTTP Method(s)

`GET`, `POST`, `DELETE`

### Example/Starter Code

**routes/users.js:**
```javascript
import express from 'express';
import { v4 as uuidv4 } from 'uuid';

const router = express.Router();

// In-memory database
let users = [];

// GET all users
router.get('/', (req, res) => {
    res.send(users);
});

// POST — create a new user
router.post('/', (req, res) => {
    const user = req.body;
    users.push({ ...user, id: uuidv4() });
    res.send(`${user.first_name} has been added to the Database`);
});

// GET user by ID
router.get('/:id', (req, res) => {
    const { id } = req.params;
    const foundUser = users.find((user) => user.id === id);
    if (!foundUser) {
        return res.status(404).send({ message: 'User not found' });
    }
    res.send(foundUser);
});

// DELETE user by ID
router.delete('/:id', (req, res) => {
    const { id } = req.params;
    const userExists = users.find((user) => user.id === id);
    if (!userExists) {
        return res.status(404).send({ message: 'User not found' });
    }
    users = users.filter((user) => user.id !== id);
    res.send(`${id} deleted successfully from database`);
});

export default router;
```

## Exercise 3: PATCH Update + Middleware Pattern for Resource Lookup

### Problem

Add a `PATCH` endpoint for partial user updates and implement a reusable **middleware** function that looks up a user by ID before the route handler runs. This eliminates duplicate lookup code across `GET`, `PATCH`, and `DELETE`:

- `PATCH /users/:id` — updates only the provided fields (`first_name`, `last_name`, `email`) on an existing user. Returns the updated user or a 404 error.
- Implement a middleware function `getUser(req, res, next)` that:
  1. Finds a user by `req.params.id` using `Array.find()`
  2. If found, attaches `req.user = foundUser` and calls `next()`
  3. If not found, returns `404 { message: 'Cannot find user' }`
- Apply `getUser` middleware to the `GET /:id`, `PATCH /:id`, and (optionally) `DELETE /:id` routes

### Concepts

- Express middleware: `(req, res, next) => { ... }`
- `next()` to pass control to the next handler in the chain
- Middleware registration: `router.get('/:id', getUser, (req, res) => {...})`
- `PATCH` vs `PUT`: `PATCH` for partial updates, `PUT` for full replacement
- Conditional field updates with `if (req.body.field != null)`
- `req.user` pattern — attaching resolved resources to the request object

### HTTP Method(s)

`PATCH`, `GET` (with middleware)

### Example/Starter Code

**routes/users.js (add middleware + PATCH to existing CRUD):**
```javascript
import express from 'express';
import { v4 as uuidv4 } from 'uuid';

const router = express.Router();
let users = [];

// ============================================================
// MIDDLEWARE: looks up a user by ID and attaches to req.user
// ============================================================
async function getUser(req, res, next) {
    const { id } = req.params;
    const foundUser = users.find((user) => user.id === id);
    if (!foundUser) {
        return res.status(404).json({ message: 'Cannot find user' });
    }
    req.user = foundUser;
    next();
}

// GET all users
router.get('/', (req, res) => {
    res.json(users);
});

// POST — create user
router.post('/', (req, res) => {
    const user = { ...req.body, id: uuidv4() };
    users.push(user);
    res.status(201).json(user);
});

// GET user by ID (uses middleware)
router.get('/:id', getUser, (req, res) => {
    res.json(req.user);
});

// PATCH — partial update (uses middleware)
router.patch('/:id', getUser, (req, res) => {
    if (req.body.first_name != null) {
        req.user.first_name = req.body.first_name;
    }
    if (req.body.last_name != null) {
        req.user.last_name = req.body.last_name;
    }
    if (req.body.email != null) {
        req.user.email = req.body.email;
    }
    res.json(req.user);
});

// DELETE — remove user (uses middleware)
router.delete('/:id', getUser, (req, res) => {
    users = users.filter((user) => user.id !== req.user.id);
    res.json({ message: 'User deleted' });
});

export default router;
```

**index.js (update to use error handling middleware):**
```javascript
import express from 'express';
import userRoutes from './routes/users.js';

const app = express();
const PORT = 5000;

app.use(express.json());
app.use('/users', userRoutes);

app.get('/', (req, res) => res.send('HELLO FROM HOMEPAGE'));

// Global error handling middleware
app.use((err, req, res, next) => {
    console.error(err.stack);
    res.status(500).json({ message: 'Internal server error' });
});

app.listen(PORT, () =>
    console.log(`Server running on port: http://localhost:${PORT}`)
);
```
