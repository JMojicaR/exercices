## Exercise 1: App Setup + GET Endpoints
**Difficulty:** Basic
**Concepts:** Express app, `app.get`, route handlers, JSON responses

Create an Express server with a health route and a route that lists all orders.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const orders = [
  { id: 1, item: "Laptop", quantity: 1, status: "pending" },
  { id: 2, item: "Mouse", quantity: 2, status: "shipped" },
];

app.get("/health", (req, res) => {
  res.json({ status: "ok" });
});

app.get("/orders", (req, res) => {
  res.json(orders);
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a `GET /orders/count` endpoint that returns `{ count: orders.length }`.

## Exercise 2: POST + Body Parsing + Validation
**Difficulty:** Basic
**Concepts:** `express.json()`, `app.post`, request body, validation, 201 status

Add an endpoint to create a new order with request-body validation.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const orders = [];
let nextId = 1;

app.post("/orders", (req, res) => {
  const { item, quantity } = req.body;
  if (!item || typeof item !== "string") {
    return res.status(400).json({ error: "item is required" });
  }
  const order = {
    id: nextId++,
    item,
    quantity: quantity || 1,
    status: "pending",
  };
  orders.push(order);
  res.status(201).json(order);
});
```

**Task:** Validate that `quantity` is a positive number; otherwise return a 400 error.

## Exercise 3: PUT/DELETE + Error Middleware
**Difficulty:** Basic
**Concepts:** `app.put`, `app.delete`, path params, 404, error-handling middleware

Add endpoints to update and delete an order, plus a catch-all error handler.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const orders = [
  { id: 1, item: "Laptop", quantity: 1, status: "pending" },
];

app.put("/orders/:id", (req, res) => {
  const order = orders.find((o) => o.id === Number(req.params.id));
  if (!order) return res.status(404).json({ error: "Order not found" });
  const { item, quantity, status } = req.body;
  if (item) order.item = item;
  if (quantity) order.quantity = quantity;
  if (status) order.status = status;
  res.json(order);
});

app.delete("/orders/:id", (req, res) => {
  const index = orders.findIndex((o) => o.id === Number(req.params.id));
  if (index === -1) return res.status(404).json({ error: "Order not found" });
  orders.splice(index, 1);
  res.json({ message: "Order deleted" });
});

app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: "Internal server error" });
});
```

**Task:** Add a 404 catch-all `app.use((req, res) => ...)` handler at the end of the routes.
