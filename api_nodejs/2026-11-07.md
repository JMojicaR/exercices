## Exercise 1: GET Endpoints + 404 Handling
**Difficulty:** Basic
**Concepts:** `express()`, `app.get()`, route params, `res.json()`, `res.status()`

Create two GET endpoints: one returning all movies, and one returning a single movie by `id` (404 if missing).

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let movies = [
  { id: 1, title: "Amores Perros", year: 2000 },
  { id: 2, title: "Y tu mamá también", year: 2001 },
  { id: 3, title: "Roma", year: 2018 },
];

app.get("/movies", (req, res) => {
  res.json(movies);
});

app.get("/movies/:id", (req, res) => {
  const movie = movies.find((m) => m.id === Number(req.params.id));
  if (!movie) {
    return res.status(404).json({ error: "Movie not found" });
  }
  res.json(movie);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a `?year=2001` query filter using `req.query.year`, and remember to `Number()` the query value before comparing. Test with `curl http://localhost:3000/movies?year=2001`.

## Exercise 2: POST + PUT (Create & Update)
**Difficulty:** Basic
**Concepts:** `app.post()`, `app.put()`, `req.body`, input validation, id generation

Add a `POST /movies` endpoint to create a movie and a `PUT /movies/:id` endpoint to replace one, both validating the request body.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let movies = [
  { id: 1, title: "Amores Perros", year: 2000 },
  { id: 2, title: "Y tu mamá también", year: 2001 },
];

app.post("/movies", (req, res) => {
  const { title, year } = req.body;
  if (!title || !year) {
    return res.status(400).json({ error: "title and year are required" });
  }
  const newMovie = {
    id: movies.length ? Math.max(...movies.map((m) => m.id)) + 1 : 1,
    title,
    year,
  };
  movies.push(newMovie);
  res.status(201).json(newMovie);
});

app.put("/movies/:id", (req, res) => {
  const movie = movies.find((m) => m.id === Number(req.params.id));
  if (!movie) {
    return res.status(404).json({ error: "Movie not found" });
  }
  const { title, year } = req.body;
  movie.title = title ?? movie.title;
  movie.year = year ?? movie.year;
  res.json(movie);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Validate that `year` is a 4-digit number and return `400` otherwise. Test with `curl -X POST -H "Content-Type: application/json" -d '{"title":"Sueño en otro idioma","year":2017}' http://localhost:3000/movies`.

## Exercise 3: DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.delete()`, 4-argument error middleware, `next()`, centralized error handling

Add a `DELETE /movies/:id` endpoint and a centralized error-handling middleware that returns a consistent JSON error shape.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let movies = [
  { id: 1, title: "Amores Perros", year: 2000 },
  { id: 2, title: "Y tu mamá también", year: 2001 },
  { id: 3, title: "Roma", year: 2018 },
];

app.delete("/movies/:id", (req, res) => {
  const index = movies.findIndex((m) => m.id === Number(req.params.id));
  if (index === -1) {
    const err = new Error("Movie not found");
    err.status = 404;
    return next(err);
  }
  const [deleted] = movies.splice(index, 1);
  res.json({ message: `Deleted '${deleted.title}'` });
});

// Error-handling middleware (must have 4 arguments)
app.use((err, req, res, next) => {
  const status = err.status || 500;
  res.status(status).json({ error: err.message || "Internal Server Error" });
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Fix the bug: `next` is referenced in the `DELETE` handler but not declared as a parameter. Add `next` to the handler's signature, then add a 404 catch-all route (`app.use((req, res) => ...)`) after all routes and verify `curl http://localhost:3000/unknown` returns JSON.
