## Exercise 1: App Setup + GET Routes (routing)
**Difficulty:** Basic
**Concepts:** `express()`, `app.get`, `res.json`, listening on a port.

Create an Express app with a `GET /health` endpoint and a `GET /users` endpoint.

**Starter code:**
```javascript
const express = require("express");
const app = express();

const users = [
  { id: 1, name: "Ana" },
  { id: 2, name: "Luis" },
];

app.get("/health", (req, res) => {
  res.json({ status: "ok" });
});

app.get("/users", (req, res) => {
  res.json(users);
});

app.listen(3000, () => {
  console.log("Server running on http://localhost:3000");
});
```

**Task:** Add a `GET /` root route that returns `{ message: "Welcome to the Users API" }`.

## Exercise 2: POST + JSON Body Parsing + Validation (request/response)
**Difficulty:** Basic
**Concepts:** `express.json()` middleware, `req.body`, POST, status codes, basic validation.

Add a `POST /users` endpoint that parses a JSON body, validates it, and creates a user.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const users = [
  { id: 1, name: "Ana" },
  { id: 2, name: "Luis" },
];
let nextId = 3;

app.get("/users", (req, res) => res.json(users));

app.post("/users", (req, res) => {
  const { name } = req.body;
  if (!name) {
    return res.status(400).json({ error: "name is required" });
  }
  const user = { id: nextId++, name };
  users.push(user);
  res.status(201).json(user);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Also accept an optional `email` field and validate that it contains an `@`.

## Exercise 3: PUT/DELETE + Error Middleware
**Difficulty:** Basic
**Concepts:** Route parameters (`:id`), `PUT`/`DELETE`, 404 handling, error-handling middleware.

Add update and delete endpoints plus a JSON 404 handler and a centralized error middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const users = [
  { id: 1, name: "Ana" },
  { id: 2, name: "Luis" },
];
let nextId = 3;

app.get("/users", (req, res) => res.json(users));

app.post("/users", (req, res) => {
  const { name } = req.body;
  if (!name) return res.status(400).json({ error: "name is required" });
  const user = { id: nextId++, name };
  users.push(user);
  res.status(201).json(user);
});

app.put("/users/:id", (req, res) => {
  const id = Number(req.params.id);
  const { name } = req.body;
  const user = users.find((u) => u.id === id);
  if (!user) return res.status(404).json({ error: "user not found" });
  user.name = name ?? user.name;
  res.json(user);
});

app.delete("/users/:id", (req, res) => {
  const id = Number(req.params.id);
  const index = users.findIndex((u) => u.id === id);
  if (index === -1) return res.status(404).json({ error: "user not found" });
  const [removed] = users.splice(index, 1);
  res.json({ deleted: removed });
});

// 404 handler
app.use((req, res) => res.status(404).json({ error: "route not found" }));

// error-handling middleware
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: "internal server error" });
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a `GET /users/:id` endpoint that returns a single user or a 404 JSON response.
