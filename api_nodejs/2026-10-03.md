## Exercise 1: App Setup & GET Routes (routing)
**Difficulty:** Basic
**Concepts:** `express()`, `app.get`, `res.json`, listening on a port.

Create an Express app with a health endpoint and a `GET /notes` endpoint that returns a list of notes.

**Starter code:**
```javascript
const express = require("express");
const app = express();

const notes = [
  { id: 1, title: "Buy milk", body: "Remember the groceries" },
  { id: 2, title: "Call dentist", body: "Book an appointment" },
];

app.get("/health", (req, res) => {
  res.json({ status: "ok" });
});

app.get("/notes", (req, res) => {
  res.json(notes);
});

app.listen(3000, () => {
  console.log("Server running on http://localhost:3000");
});
```

**Task:** Add a `GET /` root route returning `{ message: "Welcome to the Notes API" }`.

## Exercise 2: POST + GET by ID with 404 (request/response)
**Difficulty:** Basic
**Concepts:** `express.json()` middleware, `req.body`, `req.params`, `res.status`.

Add `POST /notes` to create a note and `GET /notes/:id` to fetch one, returning 404 when missing.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let notes = [
  { id: 1, title: "Buy milk", body: "Remember the groceries" },
  { id: 2, title: "Call dentist", body: "Book an appointment" },
];
let nextId = 3;

app.get("/notes", (req, res) => res.json(notes));

app.get("/notes/:id", (req, res) => {
  const id = Number(req.params.id);
  const note = notes.find((n) => n.id === id);
  if (!note) {
    return res.status(404).json({ error: "Note not found" });
  }
  res.json(note);
});

app.post("/notes", (req, res) => {
  const { title, body } = req.body;
  if (!title) {
    return res.status(400).json({ error: "title is required" });
  }
  const note = { id: nextId++, title, body: body || "" };
  notes.push(note);
  res.status(201).json(note);
});

app.listen(3000, () => console.log("Server running on http://localhost:3000"));
```

**Task:** Add basic validation that `title` is a non-empty string.

## Exercise 3: PUT/DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.put`, `app.delete`, updating/deleting items, custom error middleware.

Add update and delete endpoints plus a JSON error-handling middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let notes = [
  { id: 1, title: "Buy milk", body: "Remember the groceries" },
  { id: 2, title: "Call dentist", body: "Book an appointment" },
];
let nextId = 3;

app.put("/notes/:id", (req, res) => {
  const id = Number(req.params.id);
  const note = notes.find((n) => n.id === id);
  if (!note) {
    return res.status(404).json({ error: "Note not found" });
  }
  note.title = req.body.title ?? note.title;
  note.body = req.body.body ?? note.body;
  res.json(note);
});

app.delete("/notes/:id", (req, res) => {
  const id = Number(req.params.id);
  const index = notes.findIndex((n) => n.id === id);
  if (index === -1) {
    return res.status(404).json({ error: "Note not found" });
  }
  const removed = notes.splice(index, 1);
  res.json({ deleted: removed[0] });
});

app.use((err, req, res, next) => {
  console.error(err);
  res.status(500).json({ error: "Internal server error" });
});

app.listen(3000, () => console.log("Server running on http://localhost:3000"));
```

**Task:** Add a request-logging middleware that prints `METHOD PATH` for every request.
