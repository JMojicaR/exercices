## Exercise 1: GET Endpoints + Query Filter
**Difficulty:** Basic
**Concepts:** `express()`, `app.get`, `res.json`, route parameters, query strings

Create two GET endpoints: list all reservations, and filter by party size using a query string.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let reservations = [
  { id: 1, name: "Ana", partySize: 2, time: "19:00" },
  { id: 2, name: "Luis", partySize: 6, time: "20:30" },
];

app.get("/reservations", (req, res) => {
  const { partySize } = req.query;
  if (partySize) {
    return res.json(reservations.filter((r) => r.partySize == partySize));
  }
  res.json(reservations);
});

app.get("/reservations/:id", (req, res) => {
  const r = reservations.find((x) => x.id === Number(req.params.id));
  if (!r) {
    return res.status(404).json({ error: "Reservation not found" });
  }
  res.json(r);
});

app.listen(3000, () => console.log("Listening on http://localhost:3000"));
```

**Task:** Add a `?time=...` filter alongside `partySize`, and change the `/reservations/:id` handler to use `res.sendStatus` for the 404 case instead of a JSON body.

## Exercise 2: POST + PUT with Validation
**Difficulty:** Basic
**Concepts:** `app.post`, `app.put`, request body validation, `201` status, id generation

Add POST (create) and PUT (update) endpoints that validate the incoming body.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let reservations = [
  { id: 1, name: "Ana", partySize: 2, time: "19:00" },
  { id: 2, name: "Luis", partySize: 6, time: "20:30" },
];

app.post("/reservations", (req, res) => {
  const { name, partySize, time } = req.body;
  if (!name || !partySize || !time) {
    return res.status(400).json({ error: "name, partySize, and time are required" });
  }
  const newReservation = {
    id: Math.max(...reservations.map((r) => r.id)) + 1,
    name,
    partySize,
    time,
  };
  reservations.push(newReservation);
  res.status(201).json(newReservation);
});

app.put("/reservations/:id", (req, res) => {
  const r = reservations.find((x) => x.id === Number(req.params.id));
  if (!r) {
    return res.status(404).json({ error: "Reservation not found" });
  }
  const { name, partySize, time } = req.body;
  r.name = name ?? r.name;
  r.partySize = partySize ?? r.partySize;
  r.time = time ?? r.time;
  res.json(r);
});

app.listen(3000, () => console.log("Listening on http://localhost:3000"));
```

**Task:** Validate that `partySize` is a positive integer, returning `400` otherwise. Test with `curl -X POST -H "Content-Type: application/json" -d '{"name":"Marta","partySize":4,"time":"19:30"}' http://localhost:3000/reservations`.

## Exercise 3: DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.delete`, `app.use` error middleware, 404 fallback, `next()`

Add a DELETE endpoint, plus a final error-handling middleware and a 404 catch-all.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let reservations = [
  { id: 1, name: "Ana", partySize: 2, time: "19:00" },
  { id: 2, name: "Luis", partySize: 6, time: "20:30" },
];

app.delete("/reservations/:id", (req, res) => {
  const index = reservations.findIndex((x) => x.id === Number(req.params.id));
  if (index === -1) {
    return res.status(404).json({ error: "Reservation not found" });
  }
  const [deleted] = reservations.splice(index, 1);
  res.json({ message: `Deleted reservation for ${deleted.name}` });
});

// 404 catch-all — must be registered after all routes
app.use((req, res) => {
  res.status(404).json({ error: "Route not found" });
});

// Error-handling middleware — must have 4 arguments
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: "Internal server error" });
});

app.listen(3000, () => console.log("Listening on http://localhost:3000"));
```

**Task:** Add a route that deliberately throws (`throw new Error("boom")`) to confirm the error middleware returns 500, and verify the 404 catch-all fires for `curl http://localhost:3000/nope`. Then add a `GET /reservations/:id/time` endpoint returning just the `time` field.
