## Exercise 1: App Setup + GET Endpoints
**Difficulty:** Basic
**Concepts:** Express app, `app.get`, `res.json`, server listen

Create an Express app with a health-check route and a list endpoint.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const articles = [
  { id: 1, title: "Getting Started with Express", author: "Ana", views: 120 },
  { id: 2, title: "REST API Basics", author: "Luis", views: 95 },
];

app.get("/health", (req, res) => {
  res.json({ status: "ok" });
});

app.get("/articles", (req, res) => {
  res.json(articles);
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a `GET /articles/count` endpoint returning `{ count: articles.length }`, and log each incoming request with a simple middleware.

## Exercise 2: POST + GET by ID (validation + 404)
**Difficulty:** Basic
**Concepts:** `express.json()`, POST body, `req.params`, validation, 404

Add endpoints to create an article and fetch one by ID, with input validation.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const articles = [];
let nextId = 1;

app.post("/articles", (req, res) => {
  const { title, author } = req.body;
  if (!title) {
    return res.status(400).json({ error: "title is required" });
  }
  const article = {
    id: nextId++,
    title,
    author: author || "Anonymous",
    views: 0,
  };
  articles.push(article);
  res.status(201).json(article);
});

app.get("/articles/:id", (req, res) => {
  const id = Number(req.params.id);
  const article = articles.find((a) => a.id === id);
  if (!article) {
    return res.status(404).json({ error: "Article not found" });
  }
  res.json(article);
});

app.listen(3000);
```

**Task:** Validate that `title` is a non-empty string of at most 100 characters, returning 400 otherwise.

## Exercise 3: PUT + DELETE + Error Middleware
**Difficulty:** Basic
**Concepts:** PUT, DELETE, `find`/`findIndex`, error-handling middleware

Add update and delete endpoints plus a global error handler.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const articles = [];

app.put("/articles/:id", (req, res) => {
  const id = Number(req.params.id);
  const article = articles.find((a) => a.id === id);
  if (!article) {
    return res.status(404).json({ error: "Article not found" });
  }
  article.title = req.body.title ?? article.title;
  article.author = req.body.author ?? article.author;
  res.json(article);
});

app.delete("/articles/:id", (req, res) => {
  const id = Number(req.params.id);
  const index = articles.findIndex((a) => a.id === id);
  if (index === -1) {
    return res.status(404).json({ error: "Article not found" });
  }
  articles.splice(index, 1);
  res.json({ message: "Article deleted" });
});

// Error-handling middleware (must have 4 parameters)
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: "Internal server error" });
});

app.listen(3000);
```

**Task:** Add a 404 catch-all middleware (`app.use((req, res) => ...)`) that returns JSON for any unmatched route, placed after all route definitions.
