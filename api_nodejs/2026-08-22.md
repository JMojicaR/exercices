## Exercise 1: Hello + Route/Query Params
**Difficulty:** Basic
### Problem
Create a minimal Express server with three routes:
1. `GET /` — returns `{"message": "Hello, World!"}`.
2. `GET /hello/:name` — uses a **route parameter** to greet a name.
3. `GET /greet?name=<value>` — uses a **query parameter** (default `"Guest"`).

### Concepts
- `express()` and `app.listen`
- Route parameters (`req.params`)
- Query parameters (`req.query`)
- `res.json` for JSON responses

### HTTP Method(s)
`GET`

### Example / Starter Code
```javascript
const express = require('express');
const app = express();
const PORT = 3000;

app.get('/', (req, res) => {
  res.json({ message: 'Hello, World!' });
});

app.get('/hello/:name', (req, res) => {
  res.json({ message: `Hello, ${req.params.name}!` });
});

app.get('/greet', (req, res) => {
  const name = req.query.name || 'Guest';
  res.json({ message: `Hello, ${name}!` });
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```
**Test with curl:**
```bash
curl http://127.0.0.1:3000/
curl http://127.0.0.1:3000/hello/Alice
curl "http://127.0.0.1:3000/greet?name=Bob"
curl "http://127.0.0.1:3000/greet"
```

## Exercise 2: Movies CRUD API (in-memory)
**Difficulty:** Basic
### Problem
Build an in-memory CRUD API for movies (stored in a JavaScript array):
- `GET /movies` — list all movies.
- `POST /movies` — create a movie from JSON `{"title", "year"}` (status 201, auto-incremented `id`).
- `GET /movies/:id` — fetch one movie or 404.
- `PUT /movies/:id` — update title/year or 404.
- `DELETE /movies/:id` — remove one movie or 404.

### Concepts
- `express.json()` middleware for parsing JSON bodies
- `req.body`, `req.params`
- Array CRUD with `find`, `findIndex`, `push`, `splice`
- Correct status codes (`201`, `404`)

### HTTP Method(s)
`GET`, `POST`, `PUT`, `DELETE`

### Example / Starter Code
```javascript
const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

let movies = [];
let nextId = 1;

app.get('/movies', (req, res) => {
  res.json(movies);
});

app.post('/movies', (req, res) => {
  const movie = { id: nextId++, title: req.body.title, year: req.body.year };
  movies.push(movie);
  res.status(201).json(movie);
});

app.get('/movies/:id', (req, res) => {
  const movie = movies.find(m => m.id === parseInt(req.params.id, 10));
  if (!movie) return res.status(404).json({ error: 'Movie not found' });
  res.json(movie);
});

app.put('/movies/:id', (req, res) => {
  const movie = movies.find(m => m.id === parseInt(req.params.id, 10));
  if (!movie) return res.status(404).json({ error: 'Movie not found' });
  if (req.body.title !== undefined) movie.title = req.body.title;
  if (req.body.year !== undefined) movie.year = req.body.year;
  res.json(movie);
});

app.delete('/movies/:id', (req, res) => {
  const index = movies.findIndex(m => m.id === parseInt(req.params.id, 10));
  if (index === -1) return res.status(404).json({ error: 'Movie not found' });
  movies.splice(index, 1);
  res.json({ message: 'Movie deleted' });
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```
**Test with curl:**
```bash
curl http://127.0.0.1:3000/movies
curl -X POST http://127.0.0.1:3000/movies \
  -H "Content-Type: application/json" \
  -d '{"title": "Inception", "year": 2010}'
curl http://127.0.0.1:3000/movies/1
curl -X PUT http://127.0.0.1:3000/movies/1 \
  -H "Content-Type: application/json" \
  -d '{"year": 2011}'
curl -X DELETE http://127.0.0.1:3000/movies/1
```

## Exercise 3: Logging Middleware + 404/500 Error Handlers
**Difficulty:** Basic
### Problem
Add logging and error handling to an Express app:
1. Add a request-logging middleware that logs `METHOD url` for every request.
2. Add a 404 handler for unmatched routes (JSON).
3. Add an error-handling middleware (4-argument signature) that returns JSON for server errors.
4. Include a `/boom` route that passes an error to the handler via `next(err)`.

### Concepts
- `app.use` middleware and `next()`
- Error-handling middleware (4 arguments: `err, req, res, next`)
- `next(err)` to forward errors

### HTTP Method(s)
`GET`

### Example / Starter Code
```javascript
const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

// Request-logging middleware
app.use((req, res, next) => {
  console.log(`${req.method} ${req.url}`);
  next();
});

app.get('/', (req, res) => {
  res.json({ message: 'Welcome' });
});

app.get('/boom', (req, res, next) => {
  next(new Error('Something went wrong!'));
});

// 404 handler (no matching route)
app.use((req, res) => {
  res.status(404).json({ error: 'Not Found' });
});

// Error-handling middleware (must have 4 arguments)
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: 'Internal Server Error' });
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```
**Test with curl:**
```bash
curl http://127.0.0.1:3000/
curl http://127.0.0.1:3000/nope    # -> 404 JSON
curl http://127.0.0.1:3000/boom    # -> 500 JSON
```
