# Node.js / Express — API Exercises (2026-11-20)
Sources: Express docs, MDN Express Tutorial, freeCodeCamp

Theme: **Dictionary-Backed CRUD API** — an orders store held in an in-memory object keyed by `id`, with GET/POST/PUT/DELETE endpoints, `express.json()` middleware, validation, and centralized error handling.

Initialize with `npm init -y && npm install express`, then run `node app.js` and test at `http://127.0.0.1:3000`.

## Exercise 1: GET Endpoints (List, Filter, By ID)
**Difficulty:** Basic
**Concepts:** `express()`, `app.get`, `res.json`, route params, query strings, 404 responses

Serve the orders object, filter by `status`, and fetch one order by `id`.

**Starter code:**
```javascript
const express = require("express");
const app = express();

const orders = {
  1: { item: "Laptop", total: 1200, status: "pending" },
  2: { item: "Mouse", total: 25, status: "shipped" },
  3: { item: "Keyboard", total: 75, status: "pending" },
};

app.get("/orders", (req, res) => {
  const { status } = req.query;
  const all = Object.values(orders);
  if (status) {
    return res.json(all.filter((o) => o.status === status));
  }
  res.json(all);
});

app.get("/orders/:id", (req, res) => {
  const order = orders[req.params.id];
  if (!order) {
    return res.status(404).json({ error: "Order not found" });
  }
  res.json(order);
});

app.listen(3000, () => console.log("Listening on :3000"));
```

**Task:** Add a `?minTotal=` query filter that returns only orders whose `total` is greater than or equal to the given number.

## Exercise 2: POST & PUT (Create & Update)
**Difficulty:** Basic
**Concepts:** `express.json()`, `req.body`, `app.post`, `app.put`, validation, `201` status

Create new orders and update existing ones, validating the request body.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const orders = {
  1: { item: "Laptop", total: 1200, status: "pending" },
};

app.post("/orders", (req, res) => {
  const { item, total, status = "pending" } = req.body;
  if (!item || typeof total !== "number") {
    return res.status(400).json({ error: "item and numeric total are required" });
  }
  const id = Math.max(0, ...Object.keys(orders).map(Number)) + 1;
  orders[id] = { item, total, status };
  res.status(201).json({ id, ...orders[id] });
});

app.put("/orders/:id", (req, res) => {
  const order = orders[req.params.id];
  if (!order) {
    return res.status(404).json({ error: "Order not found" });
  }
  orders[req.params.id] = { ...order, ...req.body };
  res.json(orders[req.params.id]);
});

app.listen(3000, () => console.log("Listening on :3000"));
```

**Task:** In `PUT`, prevent callers from overwriting the `id` by deleting `req.body.id` before merging (e.g. `delete req.body.id`).

## Exercise 3: DELETE + Error-Handling Middleware + 404 Fallback
**Difficulty:** Basic
**Concepts:** `app.delete`, error-handling middleware `(err, req, res, next)`, `app.use` 404 fallback

Delete an order, add a catch-all error handler, and a 404 fallback for unknown routes.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const orders = {
  1: { item: "Laptop", total: 1200, status: "pending" },
  2: { item: "Mouse", total: 25, status: "shipped" },
};

app.delete("/orders/:id", (req, res) => {
  const { id } = req.params;
  if (!orders[id]) {
    return res.status(404).json({ error: "Order not found" });
  }
  delete orders[id];
  res.json({ deleted: Number(id), remaining: Object.keys(orders).length });
});

// 404 fallback (must be after all routes)
app.use((req, res) => {
  res.status(404).json({ error: "Route not found" });
});

// Error-handling middleware (must have 4 params)
app.use((err, req, res, next) => {
  console.error(err);
  res.status(500).json({ error: "Internal server error" });
});

app.listen(3000, () => console.log("Listening on :3000"));
```

**Task:** Trigger the 500 handler by throwing an error inside a route (e.g. `throw new Error("boom")` in a temporary `/boom` route) and confirm the middleware returns the JSON error.
