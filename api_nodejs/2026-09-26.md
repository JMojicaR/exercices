## Exercise 1: Basic Server with GET Endpoints
**Difficulty:** Basic
**Concepts:** `express()`, `app.get()`, `res.json()`, `app.listen()`.

Create an Express server with a `GET /` greeting and a `GET /health` endpoint.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.get("/", (req, res) => {
    res.json({ message: "Hello, Express API!" });
});

app.get("/health", (req, res) => {
    res.json({ status: "ok" });
});

app.listen(PORT, () => {
    console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a `GET /info` endpoint that returns the current server time and the HTTP method used.

## Exercise 2: JSON Middleware & POST/PUT Handling
**Difficulty:** Basic
**Concepts:** `express.json()` middleware, `req.body`, `req.params`, `req.query`, POST/PUT.

Create an app that parses JSON bodies and implements a `POST /echo` plus a `PUT /greet/:name` endpoint.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());

app.post("/echo", (req, res) => {
    res.json({ you_sent: req.body, method: req.method });
});

app.put("/greet/:name", (req, res) => {
    const { name } = req.params;
    const { title = "friend" } = req.query;
    res.json({ message: `Hello, ${title} ${name}!` });
});

app.listen(PORT, () => {
    console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a `GET /add?a=&b=` endpoint that parses and returns the sum of two query parameters.

## Exercise 3: In-Memory CRUD with Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** GET/POST/PUT/DELETE, `req.params`, `req.body`, status codes, error-handling middleware.

Build a "tasks" API backed by an in-memory array, with full CRUD and a central error-handling middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());

let tasks = [];
let nextId = 1;

app.get("/tasks", (req, res) => {
    res.json(tasks);
});

app.post("/tasks", (req, res) => {
    if (!req.body || !req.body.title) {
        return res.status(400).json({ error: "Missing 'title'" });
    }
    const task = { id: nextId++, title: req.body.title, done: false };
    tasks.push(task);
    res.status(201).json(task);
});

app.get("/tasks/:id", (req, res) => {
    const task = tasks.find((t) => t.id === Number(req.params.id));
    if (!task) return res.status(404).json({ error: "Task not found" });
    res.json(task);
});

app.put("/tasks/:id", (req, res) => {
    const task = tasks.find((t) => t.id === Number(req.params.id));
    if (!task) return res.status(404).json({ error: "Task not found" });
    task.title = req.body.title ?? task.title;
    task.done = req.body.done ?? task.done;
    res.json(task);
});

app.delete("/tasks/:id", (req, res) => {
    const index = tasks.findIndex((t) => t.id === Number(req.params.id));
    if (index === -1) return res.status(404).json({ error: "Task not found" });
    res.json(tasks.splice(index, 1)[0]);
});

// Central error-handling middleware (must have 4 arguments)
app.use((err, req, res, next) => {
    console.error(err.stack);
    res.status(500).json({ error: "Internal server error" });
});

app.listen(PORT, () => {
    console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a custom request-logging middleware (logging method + path + timestamp) mounted before the routes, and a `GET /tasks?done=true` filter.
