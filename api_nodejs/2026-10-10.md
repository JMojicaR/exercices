## Exercise 1: App + GET /events + GET /health
**Difficulty:** Basic
**Concepts:** Express app, routing, res.json

Create an Express app that serves a list of events as JSON plus a health-check endpoint.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const events = [
  { id: 1, name: "React Conf", date: "2026-11-12" },
  { id: 2, name: "Node Summit", date: "2026-12-03" },
];

app.get("/health", (req, res) => {
  res.json({ status: "ok" });
});

app.get("/events", (req, res) => {
  res.json(events);
});

app.listen(PORT, () => console.log(`Listening on ${PORT}`));
```

## Exercise 2: POST /events + GET /events/:id + validation
**Difficulty:** Basic
**Concepts:** express.json(), POST, route parameters, 400/404

Let clients add events and fetch one by id, validating the request body.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const events = [
  { id: 1, name: "React Conf", date: "2026-11-12" },
];

app.get("/events", (req, res) => res.json(events));

app.post("/events", (req, res) => {
  const { name, date } = req.body;
  if (!name || !date) {
    return res.status(400).json({ error: "name and date are required" });
  }
  const event = { id: events.length + 1, name, date };
  events.push(event);
  res.status(201).json(event);
});

app.get("/events/:id", (req, res) => {
  const event = events.find((e) => e.id === Number(req.params.id));
  if (!event) return res.status(404).json({ error: "event not found" });
  res.json(event);
});

app.listen(3000, () => console.log("Listening on 3000"));
```

## Exercise 3: PUT/DELETE + error-handling middleware
**Difficulty:** Basic
**Concepts:** PUT, DELETE, error middleware

Add update and delete endpoints plus a central error-handling middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const events = [
  { id: 1, name: "React Conf", date: "2026-11-12" },
];

app.get("/events", (req, res) => res.json(events));

app.post("/events", (req, res) => {
  const { name, date } = req.body;
  if (!name || !date) {
    return res.status(400).json({ error: "name and date are required" });
  }
  const event = { id: events.length + 1, name, date };
  events.push(event);
  res.status(201).json(event);
});

app.put("/events/:id", (req, res) => {
  const event = events.find((e) => e.id === Number(req.params.id));
  if (!event) return res.status(404).json({ error: "event not found" });
  Object.assign(event, req.body);
  res.json(event);
});

app.delete("/events/:id", (req, res) => {
  const index = events.findIndex((e) => e.id === Number(req.params.id));
  if (index === -1) return res.status(404).json({ error: "event not found" });
  events.splice(index, 1);
  res.json({ message: "deleted" });
});

// Error-handling middleware (must have 4 arguments)
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: "internal server error" });
});

app.listen(3000, () => console.log("Listening on 3000"));
```

**Task:** Add a catch-all 404 middleware for unknown routes that returns JSON instead of the default HTML.
