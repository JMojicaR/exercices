## Exercise 1: App Setup, Health Check & GET Collection (routing)
**Difficulty:** Basic
**Concepts:** `express()`, `app.get`, `res.json`, `app.listen`.

Create an Express server with a `GET /health` endpoint and a `GET /users` endpoint returning a list.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.get("/health", (req, res) => {
  res.json({ status: "ok" });
});

const users = [
  { id: 1, name: "Alice" },
  { id: 2, name: "Bob" },
];

app.get("/users", (req, res) => {
  res.json(users);
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a `GET /` root route that returns `{ message: "Users API" }`.

## Exercise 2: POST Create + GET by ID with Validation (request/response)
**Difficulty:** Basic
**Concepts:** `express.json()` middleware, `req.body`, `req.params`, status codes, validation.

Add a `POST /users` endpoint that validates input, and a `GET /users/:id` endpoint.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let nextId = 1;
const users = [];

app.post("/users", (req, res) => {
  const { name } = req.body;
  if (!name) {
    return res.status(400).json({ error: "name is required" });
  }
  const user = { id: nextId++, name };
  users.push(user);
  res.status(201).json(user);
});

app.get("/users/:id", (req, res) => {
  const id = parseInt(req.params.id);
  const user = users.find((u) => u.id === id);
  if (!user) {
    return res.status(404).json({ error: "User not found" });
  }
  res.json(user);
});

app.listen(3000, () => console.log("Server running on port 3000"));
```

**Task:** Add a `GET /users?name=<substring>` filter that returns users whose name contains the query string.

## Exercise 3: PUT/DELETE + Error-Handling Middleware (update/delete/errors)
**Difficulty:** Basic
**Concepts:** `app.put`, `app.delete`, `findIndex`, `splice`, error-handling middleware (`err, req, res, next`).

Add update and delete endpoints and a global error-handling middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const users = [];

app.put("/users/:id", (req, res) => {
  const id = parseInt(req.params.id);
  const user = users.find((u) => u.id === id);
  if (!user) {
    return res.status(404).json({ error: "User not found" });
  }
  user.name = req.body.name || user.name;
  res.json(user);
});

app.delete("/users/:id", (req, res) => {
  const id = parseInt(req.params.id);
  const index = users.findIndex((u) => u.id === id);
  if (index === -1) {
    return res.status(404).json({ error: "User not found" });
  }
  res.json(users.splice(index, 1)[0]);
});

// error-handling middleware (4 args)
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: "Something went wrong" });
});

app.listen(3000, () => console.log("Server running on port 3000"));
```

**Task:** Add a `POST /users` endpoint so the full CRUD cycle works, and a request-logging middleware that prints `METHOD /path` for every request.
