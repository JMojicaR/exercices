## Exercise 1: Products API — Full CRUD

**Difficulty:** Basic

### Problem
Build a REST API managing `products` with full CRUD:
- `GET /products` — list all products
- `POST /products` — create a product (`name`, `price` required); return 201 or 400
- `GET /products/:id` — return one product, or 404
- `PUT /products/:id` — update `name` and/or `price`, or 404
- `DELETE /products/:id` — delete a product, or 404

### Concepts
- `express.json()` body parsing
- Route parameters with `req.params.id`
- In-memory array store with an auto-incrementing ID
- Proper status codes with `res.status(...)`

### HTTP Method(s)
`GET`, `POST`, `PUT`, `DELETE`

### Example / Starter Code

```javascript
const express = require('express');
const app = express();
app.use(express.json());

let products = [];
let nextId = 1;

app.get('/products', (req, res) => {
    res.json(products);
});

app.post('/products', (req, res) => {
    const { name, price } = req.body;
    if (!name || price === undefined) {
        return res.status(400).json({ error: "'name' and 'price' are required" });
    }
    const product = { id: nextId++, name, price };
    products.push(product);
    res.status(201).json(product);
});

app.get('/products/:id', (req, res) => {
    const product = products.find(p => p.id === Number(req.params.id));
    if (!product) return res.status(404).json({ error: 'Product not found' });
    res.json(product);
});

app.put('/products/:id', (req, res) => {
    const product = products.find(p => p.id === Number(req.params.id));
    if (!product) return res.status(404).json({ error: 'Product not found' });
    const { name, price } = req.body;
    if (name !== undefined) product.name = name;
    if (price !== undefined) product.price = price;
    res.json(product);
});

app.delete('/products/:id', (req, res) => {
    const index = products.findIndex(p => p.id === Number(req.params.id));
    if (index === -1) return res.status(404).json({ error: 'Product not found' });
    products.splice(index, 1);
    res.json({ message: `Product ${req.params.id} deleted` });
});

app.listen(3000, () => console.log('Server on http://localhost:3000'));
```

**Test with curl:**
```bash
curl -X POST http://localhost:3000/products -H "Content-Type: application/json" -d '{"name":"Widget","price":9.99}'
curl http://localhost:3000/products
curl -X PUT http://localhost:3000/products/1 -H "Content-Type: application/json" -d '{"price":12.99}'
curl -X DELETE http://localhost:3000/products/1
```

## Exercise 2: Logging Middleware, Body Parsing & 404 Handler

**Difficulty:** Basic

### Problem
Build a small API that demonstrates the middleware pipeline:
- Use `express.json()` to parse request bodies
- Add a custom logging middleware that prints a timestamp, the method, and the path
- `GET /` — returns a welcome message
- `POST /echo` — echoes back the JSON body with status 201
- Add a final 404 handler that returns JSON for unknown routes

### Concepts
- `app.use()` for middleware registration
- `next()` to pass control down the pipeline
- Order matters: 404 handler must be registered last
- Custom logging with `console.log`

### HTTP Method(s)
`GET`, `POST`

### Example / Starter Code

```javascript
const express = require('express');
const app = express();

// Built-in JSON body parsing middleware
app.use(express.json());

// Custom logging middleware
app.use((req, res, next) => {
    console.log(`[${new Date().toISOString()}] ${req.method} ${req.path}`);
    next();
});

app.get('/', (req, res) => {
    res.send('Welcome to the API');
});

app.post('/echo', (req, res) => {
    res.status(201).json({ echo: req.body });
});

// 404 handler — must be registered last
app.use((req, res) => {
    res.status(404).json({ error: 'Route not found' });
});

app.listen(3000, () => console.log('Server on http://localhost:3000'));
```

**Test with curl:**
```bash
curl http://localhost:3000/
curl -X POST http://localhost:3000/echo -H "Content-Type: application/json" -d '{"msg":"hi"}'
curl http://localhost:3000/nope   # 404 JSON
```

## Exercise 3: Error-Handling Middleware & Query-Param Validation

**Difficulty:** Basic

### Problem
Build an `items` API with centralized error handling:
- `GET /items?limit=3` — return items, optionally limited by the `limit` query param
- `POST /items` — create an item; if `name` is missing, forward a 400 error to the error handler
- Add a 4-argument error-handling middleware `(err, req, res, next)` that returns `{ error: err.message }` with the appropriate status

### Concepts
- Query parameters via `req.query`
- Passing errors to `next(err)`
- Error-handling middleware signature `(err, req, res, next)`
- Setting a custom `err.status`

### HTTP Method(s)
`GET`, `POST`

### Example / Starter Code

```javascript
const express = require('express');
const app = express();
app.use(express.json());

const items = [];

app.get('/items', (req, res) => {
    const { limit } = req.query;
    const result = limit ? items.slice(0, Number(limit)) : items;
    res.json(result);
});

app.post('/items', (req, res, next) => {
    if (!req.body.name) {
        const err = new Error("'name' is required");
        err.status = 400;
        return next(err);
    }
    const item = { id: items.length + 1, name: req.body.name };
    items.push(item);
    res.status(201).json(item);
});

// Centralized error-handling middleware (must have 4 arguments)
app.use((err, req, res, next) => {
    const status = err.status || 500;
    res.status(status).json({ error: err.message });
});

app.listen(3000, () => console.log('Server on http://localhost:3000'));
```

**Test with curl:**
```bash
curl -X POST http://localhost:3000/items -H "Content-Type: application/json" -d '{"name":"first"}'
curl "http://localhost:3000/items?limit=1"
curl -X POST http://localhost:3000/items -H "Content-Type: application/json" -d '{}'   # 400
```
