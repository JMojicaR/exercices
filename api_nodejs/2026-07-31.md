## Exercise 1: Basic Express Server with Routes

**Title:** Your First Express API — Routes and JSON Responses

**Description:** Set up a basic Express server with multiple GET routes. Create endpoints that return JSON responses, handle route parameters, and serve a simple welcome page. Learn the Express app lifecycle.

**Difficulty:** ★★☆☆☆ (Easy)

**Concepts Practiced:**
- `express()` initialization
- Route definition (`app.get`, `app.post`)
- `res.json()` for JSON responses
- Route parameters (`:param`)
- Query string parsing (`req.query`)
- Server listening (`app.listen`)

**Starter Code:**
```javascript
// Exercise 1: Your First Express API
const express = require('express');
const app = express();
const PORT = 3000;

// Middleware to parse JSON bodies
app.use(express.json());

// Root endpoint
app.get('/', (req, res) => {
    res.json({
        message: 'Welcome to the Express API!',
        version: '1.0.0',
        endpoints: ['/hello/:name', '/greet', '/users']
    });
});

// Route with path parameter
app.get('/hello/:name', (req, res) => {
    const { name } = req.params;
    res.json({
        greeting: `Hello, ${name}!`,
        timestamp: new Date().toISOString()
    });
});

// Route with query parameters
app.get('/greet', (req, res) => {
    const name = req.query.name || 'Guest';
    const lang = req.query.lang || 'en';

    const greetings = {
        en: 'Hello',
        es: 'Hola',
        fr: 'Bonjour',
        de: 'Hallo',
    };

    res.json({
        greeting: `${greetings[lang] || greetings.en}, ${name}!`,
        language: lang
    });
});

// Multiple route methods
app.get('/users', (req, res) => {
    res.json({
        users: [
            { id: 1, name: 'Alice' },
            { id: 2, name: 'Bob' },
        ]
    });
});

app.post('/users', (req, res) => {
    const { name } = req.body;
    if (!name) {
        return res.status(400).json({ error: 'Name is required' });
    }
    res.status(201).json({
        id: 3,
        name,
        message: `User ${name} created successfully`
    });
});

app.listen(PORT, () => {
    console.log(`Server running at http://localhost:${PORT}`);
});
```


## Exercise 2: CRUD API for User Management

**Title:** User Management System — Full CRUD

**Description:** Build a complete CRUD API for managing users using in-memory storage. Implement all operations: GET all users, GET a single user by ID, POST a new user, PUT to update a user, and DELETE a user. Include proper validation and status codes.

**Difficulty:** ★★★☆☆ (Moderate)

**Concepts Practiced:**
- Full CRUD with Express
- Route parameters (`:id`)
- Request body validation
- HTTP status codes (200, 201, 400, 404)
- Array manipulation for storage
- UUID generation for IDs

**Starter Code:**
```javascript
// Exercise 2: User Management CRUD API
const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

// In-memory database
let users = [
    { id: '1', name: 'Alice Johnson', email: 'alice@example.com', role: 'admin' },
    { id: '2', name: 'Bob Smith', email: 'bob@example.com', role: 'user' },
    { id: '3', name: 'Charlie Brown', email: 'charlie@example.com', role: 'user' },
];
let nextId = 4;

// Helper: find user by ID
const findUser = (id) => users.find(u => u.id === id);

// GET all users
app.get('/users', (req, res) => {
    const { role } = req.query;
    let result = users;
    if (role) {
        result = users.filter(u => u.role === role);
    }
    res.json({ users: result, count: result.length });
});

// GET single user
app.get('/users/:id', (req, res) => {
    const user = findUser(req.params.id);
    if (!user) {
        return res.status(404).json({ error: 'User not found' });
    }
    res.json(user);
});

// POST create user
app.post('/users', (req, res) => {
    const { name, email, role } = req.body;

    // Validation
    if (!name || !email) {
        return res.status(400).json({ error: 'Name and email are required' });
    }

    // Check for duplicate email
    if (users.some(u => u.email === email)) {
        return res.status(400).json({ error: 'Email already exists' });
    }

    const user = {
        id: String(nextId++),
        name,
        email,
        role: role || 'user',
        createdAt: new Date().toISOString()
    };

    users.push(user);
    res.status(201).json(user);
});

// PUT update user
app.put('/users/:id', (req, res) => {
    const user = findUser(req.params.id);
    if (!user) {
        return res.status(404).json({ error: 'User not found' });
    }

    const { name, email, role } = req.body;

    if (name) user.name = name;
    if (email) {
        // Check duplicate (excluding current user)
        if (users.some(u => u.email === email && u.id !== user.id)) {
            return res.status(400).json({ error: 'Email already exists' });
        }
        user.email = email;
    }
    if (role) user.role = role;
    user.updatedAt = new Date().toISOString();

    res.json(user);
});

// DELETE user
app.delete('/users/:id', (req, res) => {
    const index = users.findIndex(u => u.id === req.params.id);
    if (index === -1) {
        return res.status(404).json({ error: 'User not found' });
    }

    const deleted = users.splice(index, 1)[0];
    res.json({
        message: `User '${deleted.name}' deleted`,
        remaining: users.length
    });
});

app.listen(PORT, () => {
    console.log(`User Management API at http://localhost:${PORT}`);
});
```

**Test:**
```bash

curl "http://localhost:3000/users?role=admin"
curl http://localhost:3000/users/1
curl -X POST http://localhost:3000/users -H "Content-Type: application/json" \
  -d '{"name":"Diana Prince","email":"diana@example.com"}'
curl -X PUT http://localhost:3000/users/2 -H "Content-Type: application/json" \
  -d '{"role":"admin"}'
curl -X DELETE http://localhost:3000/users/3
```

## Exercise 3: Middleware and Error Handling

**Title:** Custom Middleware — Logging, Authentication, and Error Handling

**Description:** Build an Express API with custom middleware for request logging, a simple API key authentication, and centralized error handling. Create protected routes that require an API key and public routes that don't.

**Difficulty:** ★★★☆☆ (Moderate)

**Concepts Practiced:**
- Custom middleware functions (`(req, res, next) => ...`)
- `app.use()` for global middleware
- Route-specific middleware
- Authentication via headers (`x-api-key`)
- Centralized error handling (`err, req, res, next`)
- `morgan`-style logging (manual)

**Starter Code:**
```javascript
// Exercise 3: Middleware and Error Handling
const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

// In-memory storage
const data = [];

// ===== CUSTOM MIDDLEWARE =====

// 1. Request Logger Middleware
app.use((req, res, next) => {
    const start = Date.now();
    const { method, url } = req;

    // Capture response
    const originalSend = res.send;
    res.send = function(body) {
        const duration = Date.now() - start;
        console.log(`[${new Date().toISOString()}] ${method} ${url} → ${res.statusCode} (${duration}ms)`);
        return originalSend.call(this, body);
    };

    next();
});

// 2. API Key Authentication Middleware
const API_KEYS = ['sk-test-123', 'sk-test-456'];

function requireAuth(req, res, next) {
    const apiKey = req.headers['x-api-key'];

    if (!apiKey) {
        return res.status(401).json({
            error: 'Unauthorized',
            message: 'Missing x-api-key header'
        });
    }

    if (!API_KEYS.includes(apiKey)) {
        return res.status(403).json({
            error: 'Forbidden',
            message: 'Invalid API key'
        });
    }

    // Attach user info for downstream handlers
    req.user = { apiKey, tier: apiKey === 'sk-test-123' ? 'premium' : 'basic' };
    next();
}

// 3. Rate Limiter Middleware (simple in-memory)
const requestCounts = new Map();

function rateLimiter(maxRequests = 10, windowMs = 60000) {
    return (req, res, next) => {
        const key = req.ip || 'unknown';
        const now = Date.now();

        if (!requestCounts.has(key)) {
            requestCounts.set(key, []);
        }

        const timestamps = requestCounts.get(key);
        // Remove old entries
        const recent = timestamps.filter(t => now - t < windowMs);

        if (recent.length >= maxRequests) {
            return res.status(429).json({
                error: 'Too Many Requests',
                message: `Limit: ${maxRequests} requests per ${windowMs / 1000}s`,
                retryAfter: Math.ceil((recent[0] + windowMs - now) / 1000)
            });
        }

        recent.push(now);
        requestCounts.set(key, recent);
        next();
    };
}

// ===== ROUTES =====

// Public route (no auth required)
app.get('/public', (req, res) => {
    res.json({
        message: 'This is a public endpoint',
        data: data,
        count: data.length
    });
});

// Protected route (auth required)
app.get('/protected', requireAuth, (req, res) => {
    res.json({
        message: 'This is a protected endpoint',
        user: req.user,
        data: data
    });
});

// Protected with rate limiting
app.post('/data', requireAuth, rateLimiter(5, 60000), (req, res) => {
    const { value } = req.body;
    if (!value) {
        return res.status(400).json({ error: 'Value is required' });
    }

    const entry = {
        id: data.length + 1,
        value,
        createdBy: req.user.apiKey,
        createdAt: new Date().toISOString()
    };
    data.push(entry);

    res.status(201).json(entry);
});

// Protected delete (premium only)
app.delete('/data/:id', requireAuth, (req, res, next) => {
    if (req.user.tier !== 'premium') {
        return res.status(403).json({
            error: 'Forbidden',
            message: 'Premium tier required for deletions'
        });
    }
    next(); // Continue to route handler
}, (req, res) => {
    const id = parseInt(req.params.id);
    const index = data.findIndex(d => d.id === id);

    if (index === -1) {
        return res.status(404).json({ error: 'Data not found' });
    }

    const deleted = data.splice(index, 1)[0];
    res.json({ message: `Entry ${id} deleted`, deleted });
});

// ===== CENTRALIZED ERROR HANDLING =====

// 404 handler (must be after all routes)
app.use((req, res) => {
    res.status(404).json({
        error: 'Not Found',
        message: `Route ${req.method} ${req.originalUrl} does not exist`
    });
});

// Global error handler (4 params = error handler)
app.use((err, req, res, next) => {
    console.error(`[ERROR] ${err.stack}`);

    const status = err.status || 500;
    res.status(status).json({
        error: err.name || 'Internal Server Error',
        message: err.message || 'Something went wrong',
        ...(process.env.NODE_ENV === 'development' && { stack: err.stack })
    });
});

app.listen(PORT, () => {
    console.log(`API with middleware at http://localhost:${PORT}`);
});
```
