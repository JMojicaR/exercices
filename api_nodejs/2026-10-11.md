# Node.js / Express — API Exercises (2026-10-11)
Sources: Express official docs, MDN, freeCodeCamp

**Setup:** `npm init -y` then `npm install express`, run with `node app.js`. Theme: a Todos API.

## Exercise 1: App + GET /todos
**Difficulty:** Basic
**Concepts:** Express app, routing, JSON responses

Create an Express server that returns a list of todos as JSON.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const todos = [
  { id: 1, title: "Learn Express", done: false },
  { id: 2, title: "Build an API", done: false },
];

app.get("/todos", (req, res) => {
  res.json(todos);
});

app.listen(PORT, () => console.log(`Server on http://localhost:${PORT}`));
```

## Exercise 2: POST /todos + express.json() + validation
**Difficulty:** Basic
**Concepts:** POST, body parsing, input validation, 400 responses

Accept new todos via POST and validate the request body.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());

const todos = [
  { id: 1, title: "Learn Express", done: false },
];

app.get("/todos", (req, res) => res.json(todos));

app.post("/todos", (req, res) => {
  const { title } = req.body;
  if (!title) {
    return res.status(400).json({ error: "title is required" });
  }
  const todo = { id: todos.length + 1, title, done: false };
  todos.push(todo);
  res.status(201).json(todo);
});

app.listen(PORT, () => console.log(`Server on http://localhost:${PORT}`));
```

**Task:** Add GET /todos/:id that returns a single todo or a 404 if not found.

## Exercise 3: PUT/DELETE + error-handling middleware
**Difficulty:** Basic
**Concepts:** PUT, DELETE, middleware, error handling

Add update and delete endpoints plus a catch-all error-handling middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());

const todos = [
  { id: 1, title: "Learn Express", done: false },
];

app.get("/todos", (req, res) => res.json(todos));

app.post("/todos", (req, res) => {
  const { title } = req.body;
  if (!title) return res.status(400).json({ error: "title is required" });
  const todo = { id: todos.length + 1, title, done: false };
  todos.push(todo);
  res.status(201).json(todo);
});

app.put("/todos/:id", (req, res) => {
  const todo = todos.find((t) => t.id === Number(req.params.id));
  if (!todo) return res.status(404).json({ error: "todo not found" });
  todo.title = req.body.title ?? todo.title;
  todo.done = req.body.done ?? todo.done;
  res.json(todo);
});

app.delete("/todos/:id", (req, res) => {
  const index = todos.findIndex((t) => t.id === Number(req.params.id));
  if (index === -1) return res.status(404).json({ error: "todo not found" });
  todos.splice(index, 1);
  res.status(204).send();
});

// Error-handling middleware (must have 4 params)
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: "internal server error" });
});

app.listen(PORT, () => console.log(`Server on http://localhost:${PORT}`));
```

**Task:** Add a 404 middleware (`app.use((req, res) => ...)`) that returns JSON for unmatched routes.
