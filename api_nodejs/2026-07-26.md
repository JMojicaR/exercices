## Exercise 1: CRUD API with Express Router
**Difficulty:** ★★★☆☆  
**Concepts:** Express setup, routing, middleware, JSON parsing, route parameters  

**Description:**  
Build a RESTful CRUD API for managing a movie collection using Express. Use `express.Router()` for modular route organization. Handle GET, POST, PUT, and DELETE with proper status codes.

**Example Starter Code:**
```javascript
const express = require('express');
const app = express();
const PORT = 3000;

// Middleware
app.use(express.json());

// In-memory storage
let movies = [
    { id: 1, title: 'Inception', director: 'Christopher Nolan', year: 2010 },
    { id: 2, title: 'The Matrix', director: 'The Wachowskis', year: 1999 },
    { id: 3, title: 'Interstellar', director: 'Christopher Nolan', year: 2014 },
];
let nextId = 4;

// --- Router ---
const router = express.Router();

// GET /movies — List all movies (with optional filtering)
router.get('/', (req, res) => {
    let results = [...movies];
    
    // Filter by director
    if (req.query.director) {
        results = results.filter(m =>
            m.director.toLowerCase().includes(req.query.director.toLowerCase())
        );
    }
    
    // Filter by year
    if (req.query.year) {
        results = results.filter(m => m.year === parseInt(req.query.year));
    }
    
    res.json({
        movies: results,
        count: results.length,
        total: movies.length
    });
});

// GET /movies/:id — Get a single movie
router.get('/:id', (req, res) => {
    const movie = movies.find(m => m.id === parseInt(req.params.id));
    
    if (!movie) {
        return res.status(404).json({ error: 'Movie not found' });
    }
    
    res.json(movie);
});

// POST /movies — Create a new movie
router.post('/', (req, res) => {
    const { title, director, year } = req.body;
    
    // Validation
    if (!title || !director || !year) {
        return res.status(400).json({
            error: 'Title, director, and year are required'
        });
    }
    
    const movie = {
        id: nextId++,
        title,
        director,
        year: parseInt(year)
    };
    
    movies.push(movie);
    res.status(201).json(movie);
});

// PUT /movies/:id — Update a movie
router.put('/:id', (req, res) => {
    const movie = movies.find(m => m.id === parseInt(req.params.id));
    
    if (!movie) {
        return res.status(404).json({ error: 'Movie not found' });
    }
    
    const { title, director, year } = req.body;
    
    if (title) movie.title = title;
    if (director) movie.director = director;
    if (year) movie.year = parseInt(year);
    
    res.json(movie);
});

// DELETE /movies/:id — Delete a movie
router.delete('/:id', (req, res) => {
    const index = movies.findIndex(m => m.id === parseInt(req.params.id));
    
    if (index === -1) {
        return res.status(404).json({ error: 'Movie not found' });
    }
    
    const deleted = movies.splice(index, 1)[0];
    res.json({
        message: `Movie '${deleted.title}' deleted`,
        deleted
    });
});

// Mount router
app.use('/movies', router);

// Health check
app.get('/health', (req, res) => {
    res.json({
        status: 'healthy',
        uptime: process.uptime(),
        timestamp: new Date().toISOString()
    });
});

// Start server
app.listen(PORT, () => {
    console.log(`Server running at http://localhost:${PORT}`);
    console.log(`Movies endpoint: http://localhost:${PORT}/movies`);
});

module.exports = app;  // For testing
```

**Test with curl:**
```bash

  -H "Content-Type: application/json" \
  -d '{"year": 2011}'

# Delete a movie
curl -X DELETE http://localhost:3000/movies/2
```

## Exercise 2: Middleware & Authentication
**Difficulty:** ★★★☆☆  
**Concepts:** Custom middleware, JWT-like tokens, request validation, error handling middleware  

**Description:**  
Create an API with authentication middleware that protects certain routes. Implement login/logout with token generation, a custom validation middleware for request bodies, and centralized error handling.

**Example Starter Code:**
```javascript
const express = require('express');
const crypto = require('crypto');
const app = express();
const PORT = 3000;

app.use(express.json());

// --- Data ---
const users = [
    { id: 1, username: 'admin', password: 'admin123', role: 'admin' },
    { id: 2, username: 'user', password: 'user123', role: 'user' },
];
const tokens = new Map();  // token -> user info
const items = [
    { id: 1, name: 'Laptop', price: 999 },
    { id: 2, name: 'Monitor', price: 299 },
];

// --- Custom Middleware ---

// Request logger
app.use((req, res, next) => {
    const start = Date.now();
    res.on('finish', () => {
        const duration = Date.now() - start;
        console.log(`${req.method} ${req.originalUrl} — ${res.statusCode} — ${duration}ms`);
    });
    next();
});

// Authentication middleware
function authenticate(req, res, next) {
    const authHeader = req.headers.authorization;
    
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
        return res.status(401).json({ error: 'Authentication required' });
    }
    
    const token = authHeader.split(' ')[1];
    const userInfo = tokens.get(token);
    
    if (!userInfo) {
        return res.status(401).json({ error: 'Invalid or expired token' });
    }
    
    // Check token expiry (1 hour)
    if (Date.now() - userInfo.createdAt > 3600000) {
        tokens.delete(token);
        return res.status(401).json({ error: 'Token expired' });
    }
    
    req.user = userInfo;
    next();
}

// Admin-only middleware
function requireAdmin(req, res, next) {
    if (req.user.role !== 'admin') {
        return res.status(403).json({ error: 'Admin access required' });
    }
    next();
}

// Request body validation middleware (factory)
function validateBody(requiredFields) {
    return (req, res, next) => {
        const missing = requiredFields.filter(field => !req.body[field]);
        if (missing.length > 0) {
            return res.status(400).json({
                error: `Missing required fields: ${missing.join(', ')}`
            });
        }
        next();
    };
}

// Error handling middleware
app.use((err, req, res, next) => {
    console.error('Server Error:', err.message);
    res.status(500).json({
        error: 'Internal server error',
        message: process.env.NODE_ENV === 'development' ? err.message : undefined
    });
});

// --- Routes ---

// POST /login
app.post('/login', validateBody(['username', 'password']), (req, res) => {
    const { username, password } = req.body;
    
    const user = users.find(u => u.username === username && u.password === password);
    
    if (!user) {
        return res.status(401).json({ error: 'Invalid credentials' });
    }
    
    const token = crypto.randomBytes(32).toString('hex');
    tokens.set(token, {
        id: user.id,
        username: user.username,
        role: user.role,
        createdAt: Date.now()
    });
    
    res.json({
        message: 'Login successful',
        token,
        username: user.username,
        role: user.role
    });
});

// POST /logout
app.post('/logout', authenticate, (req, res) => {
    const authHeader = req.headers.authorization;
    const token = authHeader.split(' ')[1];
    tokens.delete(token);
    res.json({ message: 'Logged out successfully' });
});

// GET /items — Public
app.get('/items', (req, res) => {
    res.json({ items, count: items.length });
});

// POST /items — Protected
app.post('/items', authenticate, validateBody(['name', 'price']), (req, res) => {
    const item = {
        id: items.length + 1,
        name: req.body.name,
        price: parseFloat(req.body.price)
    };
    items.push(item);
    res.status(201).json(item);
});

// DELETE /items/:id — Admin only
app.delete('/items/:id', authenticate, requireAdmin, (req, res) => {
    const index = items.findIndex(i => i.id === parseInt(req.params.id));
    if (index === -1) {
        return res.status(404).json({ error: 'Item not found' });
    }
    const deleted = items.splice(index, 1)[0];
    res.json({ message: `Item '${deleted.name}' deleted` });
});

// GET /me — Get current user info
app.get('/me', authenticate, (req, res) => {
    res.json({
        username: req.user.username,
        role: req.user.role,
        authenticated: true
    });
});

app.listen(PORT, () => {
    console.log(`Server running at http://localhost:${PORT}`);
});
```

## Exercise 3: Rate-Limited API with Request Validation
**Difficulty:** ★★★☆☆  
**Concepts:** Rate limiting, request validation with Joi, pagination, error handling  

**Description:**  
Build a product API with rate limiting (using a simple in-memory implementation), request validation using Joi, pagination support, and proper error response formatting.

**Example Starter Code:**
```javascript
const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

// --- Simple Rate Limiter ---
const rateLimiter = (maxRequests, windowMs) => {
    const clients = new Map();
    
    // Clean up old entries every minute
    setInterval(() => {
        const now = Date.now();
        for (const [ip, data] of clients) {
            data.requests = data.requests.filter(t => t > now - windowMs);
            if (data.requests.length === 0) clients.delete(ip);
        }
    }, 60000);
    
    return (req, res, next) => {
        const ip = req.ip || req.connection.remoteAddress;
        const now = Date.now();
        
        if (!clients.has(ip)) {
            clients.set(ip, { requests: [] });
        }
        
        const client = clients.get(ip);
        client.requests = client.requests.filter(t => t > now - windowMs);
        
        if (client.requests.length >= maxRequests) {
            const retryAfter = Math.ceil((client.requests[0] + windowMs - now) / 1000);
            return res.status(429).json({
                error: 'Too many requests',
                retryAfter: `${retryAfter} seconds`,
                limit: maxRequests,
                window: `${windowMs / 1000} seconds`
            });
        }
        
        client.requests.push(now);
        next();
    };
};

// Apply rate limiter: 10 requests per 15 seconds
app.use(rateLimiter(10, 15000));

// --- Simple Joi-like Validation ---
const validators = {
    product: (data) => {
        const errors = [];
        
        if (!data.name || typeof data.name !== 'string' || data.name.trim().length < 2) {
            errors.push('name: must be a string with at least 2 characters');
        }
        if (data.price === undefined || typeof data.price !== 'number' || data.price <= 0) {
            errors.push('price: must be a positive number');
        }
        if (data.category && !['electronics', 'clothing', 'books', 'food'].includes(data.category)) {
            errors.push('category: must be one of [electronics, clothing, books, food]');
        }
        if (data.rating !== undefined && (data.rating < 0 || data.rating > 5)) {
            errors.push('rating: must be between 0 and 5');
        }
        
        return errors;
    }
};

// --- Data Store ---
const products = [
    { id: 1, name: 'Laptop Pro', price: 1299.99, category: 'electronics', rating: 4.5 },
    { id: 2, name: 'Running Shoes', price: 89.99, category: 'clothing', rating: 4.2 },
    { id: 3, name: 'Coffee Maker', price: 49.99, category: 'electronics', rating: 4.0 },
    { id: 4, name: 'Python Book', price: 39.99, category: 'books', rating: 4.8 },
    { id: 5, name: 'Organic Pasta', price: 3.99, category: 'food', rating: 4.1 },
];
let nextId = 6;

// --- Routes ---

// GET /products — List with pagination and filters
app.get('/products', (req, res) => {
    let results = [...products];
    
    // Filtering
    if (req.query.category) {
        results = results.filter(p => p.category === req.query.category);
    }
    if (req.query.minPrice) {
        results = results.filter(p => p.price >= parseFloat(req.query.minPrice));
    }
    if (req.query.maxPrice) {
        results = results.filter(p => p.price <= parseFloat(req.query.maxPrice));
    }
    if (req.query.search) {
        const term = req.query.search.toLowerCase();
        results = results.filter(p => p.name.toLowerCase().includes(term));
    }
    
    // Sorting
    if (req.query.sortBy) {
        const field = req.query.sortBy;
        const order = req.query.order === 'desc' ? -1 : 1;
        results.sort((a, b) => (a[field] - b[field]) * order);
    }
    
    // Pagination
    const page = parseInt(req.query.page) || 1;
    const limit = parseInt(req.query.limit) || 10;
    const startIndex = (page - 1) * limit;
    const endIndex = startIndex + limit;
    
    const paginated = results.slice(startIndex, endIndex);
    
    res.json({
        products: paginated,
        pagination: {
            page,
            limit,
            total: results.length,
            pages: Math.ceil(results.length / limit),
            hasNext: endIndex < results.length,
            hasPrev: page > 1
        }
    });
});

// GET /products/:id
app.get('/products/:id', (req, res) => {
    const product = products.find(p => p.id === parseInt(req.params.id));
    if (!product) {
        return res.status(404).json({ error: 'Product not found' });
    }
    res.json(product);
});

// POST /products — with validation
app.post('/products', (req, res) => {
    const errors = validators.product(req.body);
    
    if (errors.length > 0) {
        return res.status(400).json({
            error: 'Validation failed',
            details: errors
        });
    }
    
    const product = {
        id: nextId++,
        name: req.body.name.trim(),
        price: req.body.price,
        category: req.body.category || 'electronics',
        rating: req.body.rating || 0
    };
    
    products.push(product);
    res.status(201).json(product);
});

// GET /categories — List unique categories with count
app.get('/categories', (req, res) => {
    const categoryCounts = {};
    
    for (const product of products) {
        categoryCounts[product.category] = (categoryCounts[product.category] || 0) + 1;
    }
    
    const categories = Object.entries(categoryCounts).map(([name, count]) => ({
        name,
        count
    }));
    
    res.json({ categories });
});

// GET /stats — Basic statistics
app.get('/stats', (req, res) => {
    const prices = products.map(p => p.price);
    const ratings = products.map(p => p.rating);
    
    res.json({
        totalProducts: products.length,
        averagePrice: (prices.reduce((a, b) => a + b, 0) / prices.length).toFixed(2),
        minPrice: Math.min(...prices).toFixed(2),
        maxPrice: Math.max(...prices).toFixed(2),
        averageRating: (ratings.reduce((a, b) => a + b, 0) / ratings.length).toFixed(1)
    });
});

// Error handling for unmatched routes
app.use((req, res) => {
    res.status(404).json({
        error: 'Route not found',
        path: req.originalUrl,
        method: req.method
    });
});

app.listen(PORT, () => {
    console.log(`Products API running at http://localhost:${PORT}`);
});
```
