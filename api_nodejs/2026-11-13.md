## Exercise 1: GET Endpoints + Query Filtering
**Difficulty:** Basic
**Concepts:** `express()` app, `app.get`, `res.json`, route params, `req.query`

Create two GET endpoints: one listing all courses (with an optional `?level=` filter), and one returning a single course by `id` (404 if missing).

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const courses = [
  { id: 1, title: "Intro to Python", level: "Beginner", hours: 20 },
  { id: 2, title: "Node.js Basics", level: "Beginner", hours: 15 },
  { id: 3, title: "Advanced Rust", level: "Advanced", hours: 40 },
];

app.get("/courses", (req, res) => {
  const { level } = req.query;
  if (level) {
    return res.json(courses.filter((c) => c.level === level));
  }
  res.json(courses);
});

app.get("/courses/:id", (req, res) => {
  const course = courses.find((c) => c.id === Number(req.params.id));
  if (!course) {
    return res.status(404).json({ error: "Course not found" });
  }
  res.json(course);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a `?hours=` filter that returns courses with `hours` greater than or equal to the given value. Test with `curl "http://localhost:3000/courses?hours=20"`.

## Exercise 2: POST + PUT (Create & Update)
**Difficulty:** Basic
**Concepts:** `app.post`, `app.put`, `req.body`, input validation, status codes

Add a `POST /courses` endpoint to create a course and a `PUT /courses/:id` endpoint to update one, both validating the request body.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const courses = [
  { id: 1, title: "Intro to Python", level: "Beginner", hours: 20 },
  { id: 2, title: "Node.js Basics", level: "Beginner", hours: 15 },
];

app.post("/courses", (req, res) => {
  const { title, level, hours } = req.body;
  if (!title || !level || hours === undefined) {
    return res.status(400).json({ error: "title, level and hours are required" });
  }
  const course = {
    id: Math.max(...courses.map((c) => c.id)) + 1,
    title,
    level,
    hours,
  };
  courses.push(course);
  res.status(201).json(course);
});

app.put("/courses/:id", (req, res) => {
  const course = courses.find((c) => c.id === Number(req.params.id));
  if (!course) {
    return res.status(404).json({ error: "Course not found" });
  }
  course.title = req.body.title ?? course.title;
  course.level = req.body.level ?? course.level;
  course.hours = req.body.hours ?? course.hours;
  res.json(course);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a `PATCH /courses/:id` endpoint that updates only the supplied fields. Test `POST` with `curl -X POST -H "Content-Type: application/json" -d '{"title":"SQL","level":"Beginner","hours":12}' http://localhost:3000/courses`.

## Exercise 3: DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.delete`, `app.use` error middleware, `next(err)`, 404 handling

Add a `DELETE /courses/:id` endpoint and a centralized error-handling middleware that returns a consistent JSON error shape for any thrown error.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const courses = [
  { id: 1, title: "Intro to Python", level: "Beginner", hours: 20 },
  { id: 2, title: "Node.js Basics", level: "Beginner", hours: 15 },
];

app.delete("/courses/:id", (req, res) => {
  const index = courses.findIndex((c) => c.id === Number(req.params.id));
  if (index === -1) {
    return res.status(404).json({ error: "Course not found" });
  }
  const [deleted] = courses.splice(index, 1);
  res.json({ message: `Deleted '${deleted.title}'` });
});

// Error-handling middleware (must have 4 params)
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: "Internal server error" });
});

// 404 fallback for unknown routes
app.use((req, res) => {
  res.status(404).json({ error: "Route not found" });
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a route that intentionally throws an error (`throw new Error("boom")`) and verify the error middleware catches it. Then add a `GET /courses/count` endpoint that returns `{ total: courses.length }`.
