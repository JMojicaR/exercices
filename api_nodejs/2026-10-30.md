## Exercise 1: App Setup + GET Endpoints
**Difficulty:** Basic
**Concepts:** `express()`, `app.listen`, `app.get`, route parameters, `res.json`

Create an Express app with an in-memory list of workouts and two GET endpoints: all workouts and one by `id` (404 if missing).

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const workouts = [
  { id: 1, name: "Morning Run", type: "cardio", minutes: 30 },
  { id: 2, name: "Push Day", type: "strength", minutes: 45 },
];

app.get("/workouts", (req, res) => {
  res.json(workouts);
});

app.get("/workouts/:id", (req, res) => {
  const workout = workouts.find((w) => w.id === Number(req.params.id));
  if (!workout) {
    return res.status(404).json({ error: "Workout not found" });
  }
  res.json(workout);
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a `GET /workouts?type=cardio` query filter using `req.query.type`.

## Exercise 2: POST + PUT (Create & Update)
**Difficulty:** Basic
**Concepts:** `express.json()` middleware, `req.body`, validation, status codes, id generation

Add `POST /workouts` (with body validation) and `PUT /workouts/:id` endpoints.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const workouts = [
  { id: 1, name: "Morning Run", type: "cardio", minutes: 30 },
];

app.post("/workouts", (req, res) => {
  const { name, type, minutes } = req.body;
  if (!name || !type || typeof minutes !== "number") {
    return res.status(400).json({ error: "name, type, and minutes are required" });
  }
  const newWorkout = {
    id: workouts.length ? Math.max(...workouts.map((w) => w.id)) + 1 : 1,
    name,
    type,
    minutes,
  };
  workouts.push(newWorkout);
  res.status(201).json(newWorkout);
});

app.put("/workouts/:id", (req, res) => {
  const workout = workouts.find((w) => w.id === Number(req.params.id));
  if (!workout) {
    return res.status(404).json({ error: "Workout not found" });
  }
  const { name, type, minutes } = req.body;
  if (name !== undefined) workout.name = name;
  if (type !== undefined) workout.type = type;
  if (minutes !== undefined) workout.minutes = minutes;
  res.json(workout);
});

app.listen(3000, () => console.log("Server running on port 3000"));
```

**Task:** Test with `curl` or Postman, and make `minutes` a required positive integer (reject negative or non-numeric values).

## Exercise 3: DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.delete`, error-handling middleware (`(err, req, res, next)`), 404 handler

Add a `DELETE /workouts/:id` endpoint and a global error-handling + 404 middleware chain.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const workouts = [
  { id: 1, name: "Morning Run", type: "cardio", minutes: 30 },
  { id: 2, name: "Push Day", type: "strength", minutes: 45 },
];

app.delete("/workouts/:id", (req, res) => {
  const index = workouts.findIndex((w) => w.id === Number(req.params.id));
  if (index === -1) {
    return res.status(404).json({ error: "Workout not found" });
  }
  const [removed] = workouts.splice(index, 1);
  res.json({ message: "Deleted", removed });
});

// 404 handler — must be registered after all routes
app.use((req, res) => {
  res.status(404).json({ error: "Route not found" });
});

// Error-handling middleware — must have 4 parameters
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: "Something went wrong" });
});

app.listen(3000, () => console.log("Server running on port 3000"));
```

**Task:** Throw an error inside a route (e.g. `throw new Error("boom")`) and confirm the error-handling middleware returns a 500 JSON response.
