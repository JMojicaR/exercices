## 1. Hello World GET Endpoint
**Description:** Create a minimal Express server with a `GET /` route returning a JSON greeting and a `GET /hello/:name` route demonstrating route parameters.
**Difficulty:** Basic
**Concepts:** `express()`, `app.get`, route params (`req.params`), `res.json`, `app.listen`.
**Starter code:**
```javascript
const express = require("express");
const app = express();

app.get("/", (req, res) => {
  res.json({ message: "Hello, World!" });
});

app.get("/hello/:name", (req, res) => {
  res.json({ message: `Hello, ${req.params.name}!` });
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

## 2. In-Memory CRUD API (Todos)
**Description:** Build a full CRUD API for a "todo" resource stored in an in-memory array. Implement GET (list/single), POST, PUT, and DELETE.
**Difficulty:** Basic
**Concepts:** `express.json()` middleware, `req.body`, route params, HTTP status codes (`res.status`).
**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let todos = [];
let nextId = 1;

app.get("/todos", (req, res) => res.json(todos));

app.post("/todos", (req, res) => {
  const todo = { id: nextId++, title: req.body.title, done: false };
  todos.push(todo);
  res.status(201).json(todo);
});

app.get("/todos/:id", (req, res) => {
  const todo = todos.find(t => t.id === Number(req.params.id));
  if (!todo) return res.status(404).json({ error: "Not found" });
  res.json(todo);
});

app.put("/todos/:id", (req, res) => {
  const todo = todos.find(t => t.id === Number(req.params.id));
  if (!todo) return res.status(404).json({ error: "Not found" });
  todo.title = req.body.title ?? todo.title;
  todo.done = req.body.done ?? todo.done;
  res.json(todo);
});

app.delete("/todos/:id", (req, res) => {
  todos = todos.filter(t => t.id !== Number(req.params.id));
  res.status(204).send();
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

## 3. Middleware and Error Handling
**Description:** Add a custom logging middleware and a centralized error handler. Validate POST input and return proper 400/404 responses when data is missing or routes don't exist.
**Difficulty:** Basic
**Concepts:** Custom middleware (`req, res, next`), `next()`, error-handling middleware (4-arg signature), `app.use` ordering.
**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

// Logging middleware
app.use((req, res, next) => {
  console.log(`${req.method} ${req.url}`);
  next();
});

const items = [];

app.post("/items", (req, res, next) => {
  if (!req.body || !req.body.name) {
    return next(new Error("Field 'name' is required"));
  }
  items.push(req.body.name);
  res.status(201).json({ added: req.body.name });
});

// 404 handler for unknown routes
app.use((req, res) => {
  res.status(404).json({ error: "Resource not found" });
});

// Centralized error handler (must have 4 args)
app.use((err, req, res, next) => {
  res.status(400).json({ error: err.message });
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```
