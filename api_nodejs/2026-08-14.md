## Exercise 1: Users API — Basic Server with GET and POST

### Problem

Build an Express server that manages users with in-memory storage. Create two endpoints:

- `GET /users` — returns the full list of users as JSON
- `POST /users` — accepts a JSON body (`name`, `email`), assigns an `id`, and returns the created user with status 201

### Concepts

- `express()` app creation and `app.listen()`
- `express.json()` middleware for parsing JSON bodies
- Route handlers with `req` and `res`
- `res.json()` and `res.status()`

### HTTP Method(s)

`GET`, `POST`

### Example/Starter Code

```javascript
const express = require('express');
const app = express();

app.use(express.json());

const users = [];
let nextId = 1;

app.get('/users', (req, res) => {
  res.json(users);
});

app.post('/users', (req, res) => {
  const { name, email } = req.body;
  if (!name || !email) {
    return res.status(400).json({ error: "Both 'name' and 'email' are required" });
  }
  const user = { id: nextId++, name, email };
  users.push(user);
  res.status(201).json(user);
});

app.listen(3000, () => {
  console.log('Server running on http://localhost:3000');
});
```

**Test:**
```bash
node server.js
curl http://localhost:3000/users
curl -X POST http://localhost:3000/users \
  -H "Content-Type: application/json" \
  -d '{"name":"Alice","email":"alice@example.com"}'
```

## Exercise 2: Tasks API — Full CRUD with 404 Handling

### Problem

Build a full CRUD API for `tasks` with proper 404 handling:

- `GET /tasks` — returns all tasks
- `POST /tasks` — creates a task (`title`, `done`)
- `GET /tasks/:id` — returns one task, or 404 if not found
- `PUT /tasks/:id` — updates a task, or 404 if not found
- `DELETE /tasks/:id` — deletes a task, or 404 if not found

### Concepts

- Route parameters (`:id`)
- `req.params.id` and `parseInt()`
- `Array.find()` and `Array.findIndex()`
- Status codes `200`, `201`, `404`
- In-memory array store

### HTTP Method(s)

`GET`, `POST`, `PUT`, `DELETE`

### Example/Starter Code

```javascript
const express = require('express');
const app = express();

app.use(express.json());

const tasks = [];
let nextId = 1;

app.get('/tasks', (req, res) => res.json(tasks));

app.post('/tasks', (req, res) => {
  const { title } = req.body;
  if (!title) {
    return res.status(400).json({ error: "'title' is required" });
  }
  const task = { id: nextId++, title, done: req.body.done ?? false };
  tasks.push(task);
  res.status(201).json(task);
});

app.get('/tasks/:id', (req, res) => {
  const task = tasks.find(t => t.id === parseInt(req.params.id));
  if (!task) return res.status(404).json({ error: 'Task not found' });
  res.json(task);
});

app.put('/tasks/:id', (req, res) => {
  const task = tasks.find(t => t.id === parseInt(req.params.id));
  if (!task) return res.status(404).json({ error: 'Task not found' });
  if (req.body.title !== undefined) task.title = req.body.title;
  if (req.body.done !== undefined) task.done = req.body.done;
  res.json(task);
});

app.delete('/tasks/:id', (req, res) => {
  const index = tasks.findIndex(t => t.id === parseInt(req.params.id));
  if (index === -1) return res.status(404).json({ error: 'Task not found' });
  tasks.splice(index, 1);
  res.json({ message: `Task ${req.params.id} deleted` });
});

app.listen(3000, () => console.log('Server on http://localhost:3000'));
```

**Test:**
```bash
curl -X POST http://localhost:3000/tasks \
  -H "Content-Type: application/json" \
  -d '{"title":"Learn Express"}'
curl http://localhost:3000/tasks/1
curl -X DELETE http://localhost:3000/tasks/1
```

## Exercise 3: Logging Middleware and Error Handling

### Problem

Build a products API that adds custom middleware and error handling:

- `GET /products` — returns products, optionally filtered by `category` query parameter (e.g. `/products?category=books`)
- Add a logging middleware that prints the HTTP method and path of every request
- Add a 404 handler (`app.use`) that returns JSON for unmatched routes
- Add a 4-argument error-handling middleware that catches errors and returns a JSON 500 response

### Concepts

- `app.use()` for middleware
- Custom logging middleware with `next()`
- Reading query parameters (`req.query`)
- Error-handling middleware (4 arguments: `err, req, res, next`)
- Catch-all 404 handler

### HTTP Method(s)

`GET`

### Example/Starter Code

```javascript
const express = require('express');
const app = express();

app.use(express.json());

// Logging middleware
app.use((req, res, next) => {
  console.log(`[LOG] ${req.method} ${req.path}`);
  next();
});

const products = [
  { id: 1, name: 'Clean Code', category: 'books', price: 29.99 },
  { id: 2, name: 'Desk Lamp', category: 'home', price: 15.50 },
  { id: 3, name: 'Atomic Habits', category: 'books', price: 18.75 },
];

app.get('/products', (req, res) => {
  const { category } = req.query;
  let result = products;
  if (category) {
    result = products.filter(p => p.category === category);
  }
  res.json(result);
});

// Catch-all 404 handler
app.use((req, res) => {
  res.status(404).json({ error: 'Route not found' });
});

// Error-handling middleware (must have 4 arguments)
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: 'Internal server error' });
});

app.listen(3000, () => console.log('Server on http://localhost:3000'));
```

**Test:**
```bash
curl "http://localhost:3000/products?category=books"
curl http://localhost:3000/unknown   # 404 JSON
```
