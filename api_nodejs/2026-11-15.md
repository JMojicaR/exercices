## Exercise 1: GET Endpoints + Query Filters
**Difficulty:** Basic
**Concepts:** `express`, `app.get`, route params, `req.query`, `res.status().json()`

Create GET endpoints for all orders and for a single order, with a `?status=...` filter.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let orders = [
  { id: 1, customer: "Ana", total: 99.5, status: "pending" },
  { id: 2, customer: "Luis", total: 245.0, status: "shipped" },
];

app.get("/orders", (req, res) => {
  const { status } = req.query;
  let result = orders;
  if (status) {
    result = orders.filter((o) => o.status === status);
  }
  res.json(result);
});

app.get("/orders/:id", (req, res) => {
  const order = orders.find((o) => o.id === Number(req.params.id));
  if (!order) {
    return res.status(404).json({ error: "Order not found" });
  }
  res.json(order);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a `?minTotal=...` query filter to `GET /orders`, and test with `curl "http://localhost:3000/orders?status=shipped"`.

## Exercise 2: POST + PUT with Validation
**Difficulty:** Basic
**Concepts:** `app.post`, `app.put`, `req.body`, manual validation, `201` status, id generation

Add `POST /orders` to create and `PUT /orders/:id` to update, both validating the request body.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let orders = [
  { id: 1, customer: "Ana", total: 99.5, status: "pending" },
  { id: 2, customer: "Luis", total: 245.0, status: "shipped" },
];

app.post("/orders", (req, res) => {
  const { customer, total, status = "pending" } = req.body;
  if (!customer || total === undefined) {
    return res.status(400).json({ error: "customer and total are required" });
  }
  const newOrder = {
    id: Math.max(...orders.map((o) => o.id)) + 1,
    customer,
    total,
    status,
  };
  orders.push(newOrder);
  res.status(201).json(newOrder);
});

app.put("/orders/:id", (req, res) => {
  const order = orders.find((o) => o.id === Number(req.params.id));
  if (!order) {
    return res.status(404).json({ error: "Order not found" });
  }
  order.customer = req.body.customer ?? order.customer;
  order.total = req.body.total ?? order.total;
  order.status = req.body.status ?? order.status;
  res.json(order);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Validate that `total` is a non-negative number, returning `400` if not. Test with `curl -X POST -H "Content-Type: application/json" -d '{"customer":"Mia","total":180}' http://localhost:3000/orders`.

## Exercise 3: DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.delete`, error-handling middleware (4-arg `(err, req, res, next)`), a 404 fallback

Add `DELETE /orders/:id` plus a catch-all 404 handler and centralized error middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let orders = [
  { id: 1, customer: "Ana", total: 99.5, status: "pending" },
  { id: 2, customer: "Luis", total: 245.0, status: "shipped" },
];

app.delete("/orders/:id", (req, res) => {
  const index = orders.findIndex((o) => o.id === Number(req.params.id));
  if (index === -1) {
    return res.status(404).json({ error: "Order not found" });
  }
  const [deleted] = orders.splice(index, 1);
  res.json({ message: `Deleted order ${deleted.id}`, order: deleted });
});

// 404 fallback for any unmatched route
app.use((req, res) => {
  res.status(404).json({ error: "Route not found" });
});

// Centralized error-handling middleware (must have 4 args)
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: "Internal server error" });
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a test route that calls `next(new Error("boom"))` to confirm the error middleware returns a JSON 500, and verify the 404 fallback with `curl http://localhost:3000/nope`.
