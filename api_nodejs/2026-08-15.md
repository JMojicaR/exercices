## Exercise 1: Basic Server & Routes

**Difficulty:** Basic

### Problem
Create an Express server with:
- `GET /` — returns a welcome text
- `GET /api/greeting` — returns JSON `{"message": "Hello, World!"}`
- `POST /api/greeting` — accepts a JSON body `{ "name": "..." }` and returns a personalized greeting

### Concepts
- `express()` app initialization
- `app.listen()` to start the server
- `app.get()` / `app.post()` route handlers
- `express.json()` body-parser middleware
- `res.json()` and `res.status()`

### HTTP Method(s)
`GET`, `POST`

### Example / Starter Code

```javascript
const express = require('express');
const app = express();
const PORT = 3000;

app.use(express.json());

app.get('/', (req, res) => {
  res.send('Welcome to the Express API!');
});

app.get('/api/greeting', (req, res) => {
  res.json({ message: 'Hello, World!' });
});

app.post('/api/greeting', (req, res) => {
  const { name } = req.body;
  if (!name) {
    return res.status(400).json({ error: "'name' is required" });
  }
  res.status(201).json({ message: `Hello, ${name}!` });
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

**Test with curl:**
```bash
curl http://localhost:3000/api/greeting
curl -X POST http://localhost:3000/api/greeting \
  -H "Content-Type: application/json" -d '{"name":"Ada"}'
```

## Exercise 2: Tasks API — Full CRUD with 404 Handling

**Difficulty:** Basic

### Problem
Build a full CRUD API for `tasks` using an in-memory array:
- `GET /tasks` — list all tasks
- `POST /tasks` — create a task (`title`, `done` default `false`)
- `GET /tasks/:id` — fetch one task, or 404
- `PUT /tasks/:id` — update a task, or 404
- `DELETE /tasks/:id` — delete a task, or 404

### Concepts
- Route parameters (`req.params.id`)
- Array methods (`find`, `findIndex`, `splice`)
- `res.status(404).json(...)` for missing resources
- Auto-incrementing ID pattern

### HTTP Method(s)
`GET`, `POST`, `PUT`, `DELETE`

### Example / Starter Code

```javascript
const express = require('express');
const app = express();
app.use(express.json());

let tasks = [];
let nextId = 1;

app.get('/tasks', (req, res) => {
  res.json(tasks);
});

app.post('/tasks', (req, res) => {
  const { title } = req.body;
  if (!title) {
    return res.status(400).json({ error: "'title' is required" });
  }
  const task = { id: nextId++, title, done: req.body.done ?? false };
  tasks.push(task);
  res.status(201).json(task);
});

app.get('/tasks/:id', (req, res) => {
  const task = tasks.find((t) => t.id === Number(req.params.id));
  if (!task) {
    return res.status(404).json({ error: 'Task not found' });
  }
  res.json(task);
});

app.put('/tasks/:id', (req, res) => {
  const task = tasks.find((t) => t.id === Number(req.params.id));
  if (!task) {
    return res.status(404).json({ error: 'Task not found' });
  }
  if (req.body.title !== undefined) task.title = req.body.title;
  if (req.body.done !== undefined) task.done = req.body.done;
  res.json(task);
});

app.delete('/tasks/:id', (req, res) => {
  const index = tasks.findIndex((t) => t.id === Number(req.params.id));
  if (index === -1) {
    return res.status(404).json({ error: 'Task not found' });
  }
  tasks.splice(index, 1);
  res.json({ message: `Task ${req.params.id} deleted` });
});

app.listen(3000, () => console.log('Server on http://localhost:3000'));
```

**Test with curl:**
```bash
curl -X POST http://localhost:3000/tasks \
  -H "Content-Type: application/json" -d '{"title":"Do laundry"}'
curl http://localhost:3000/tasks
curl -X DELETE http://localhost:3000/tasks/1
```

## Exercise 3: Middleware & Error-Handling Middleware

**Difficulty:** Basic

### Problem
Extend a `products` API with custom middleware and a centralized error handler:
- `GET /products` — list all products
- `POST /products` — create a product (`name`, `price`); throws an error if `price` is negative
- Custom logging middleware that prints the method and URL of each request
- Error-handling middleware (`err, req, res, next`) that returns a consistent JSON error with the correct status code

### Concepts
- `app.use()` middleware ordering
- `next()` to pass control down the chain
- Custom `Error` objects with a `status` property
- Four-argument error-handling middleware signature

### HTTP Method(s)
`GET`, `POST`

### Example / Starter Code

```javascript
const express = require('express');
const app = express();
app.use(express.json());

const products = [];
let nextId = 1;

// Logging middleware
app.use((req, res, next) => {
  console.log(`[LOG] ${req.method} ${req.url}`);
  next();
});

app.get('/products', (req, res) => {
  res.json(products);
});

app.post('/products', (req, res, next) => {
  const { name, price } = req.body;
  if (!name || price === undefined) {
    return next(Object.assign(new Error("'name' and 'price' are required"), { status: 400 }));
  }
  if (price < 0) {
    return next(Object.assign(new Error("'price' cannot be negative"), { status: 400 }));
  }
  const product = { id: nextId++, name, price };
  products.push(product);
  res.status(201).json(product);
});

// 404 handler
app.use((req, res) => {
  res.status(404).json({ error: 'Route not found' });
});

// Centralized error-handling middleware
app.use((err, req, res, next) => {
  const status = err.status || 500;
  res.status(status).json({ error: err.message });
});

app.listen(3000, () => console.log('Server on http://localhost:3000'));
```

**Test with curl:**
```bash
curl -X POST http://localhost:3000/products \
  -H "Content-Type: application/json" -d '{"name":"Desk","price":49.99}'
curl -X POST http://localhost:3000/products \
  -H "Content-Type: application/json" -d '{"name":"Bad","price":-5}'   # 400
curl http://localhost:3000/nonexistent                                  # 404
```
