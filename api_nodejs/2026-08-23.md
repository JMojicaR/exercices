## Exercise 1: Math Utility API (route + query params)
**Difficulty:** Basic
### Problem
Build a math utility API:
1. `GET /factorial/:n` — return the factorial of `n` (route parameter).
2. `GET /fibonacci?n=<int>` — return the first `n` Fibonacci numbers (query parameter).
3. `GET /is-prime?n=<int>` — return whether `n` is prime.
Return a JSON error for invalid input.

### Concepts
- `app.get()` routes
- `req.params` for route parameters
- `req.query` for query parameters
- `res.json()` for JSON responses

### HTTP Method(s)
`GET`

### Example / Starter Code
```javascript
const express = require('express');
const app = express();

const factorial = (n) => {
  let result = 1;
  for (let i = 1; i <= n; i++) result *= i;
  return result;
};

const fibonacci = (n) => {
  const seq = [];
  let a = 0, b = 1;
  for (let i = 0; i < n; i++) {
    seq.push(a);
    [a, b] = [b, a + b];
  }
  return seq;
};

const isPrime = (n) => {
  if (n < 2) return false;
  for (let i = 2; i <= Math.sqrt(n); i++) {
    if (n % i === 0) return false;
  }
  return true;
};

app.get('/factorial/:n', (req, res) => {
  const n = parseInt(req.params.n, 10);
  if (n < 0) return res.status(400).json({ error: 'n must be non-negative' });
  res.json({ n, factorial: factorial(n) });
});

app.get('/fibonacci', (req, res) => {
  const n = parseInt(req.query.n, 10) || 0;
  if (n < 0) return res.status(400).json({ error: 'n must be non-negative' });
  res.json({ n, fibonacci: fibonacci(n) });
});

app.get('/is-prime', (req, res) => {
  const n = parseInt(req.query.n, 10);
  if (isNaN(n)) return res.status(400).json({ error: 'n must be an integer' });
  res.json({ n, isPrime: isPrime(n) });
});

app.listen(3000, () => console.log('Server on http://127.0.0.1:3000'));
```
**Test with curl:**
```bash
curl "http://127.0.0.1:3000/factorial/5"
curl "http://127.0.0.1:3000/fibonacci?n=10"
curl "http://127.0.0.1:3000/is-prime?n=17"
```

## Exercise 2: Todo CRUD API (in-memory)
**Difficulty:** Basic
### Problem
Build an in-memory CRUD API for todos:
- `GET /todos` — list all todos.
- `POST /todos` — create a todo from JSON `{"title", "done"}` (auto-increment `id`).
- `GET /todos/:id` — return one todo or 404.
- `PUT /todos/:id` — update a todo or 404.
- `DELETE /todos/:id` — remove a todo or 404.

### Concepts
- `express.json()` middleware
- Array in-memory storage with auto-incrementing id
- Status codes (`201` on create, `404` on missing)

### HTTP Method(s)
`GET`, `POST`, `PUT`, `DELETE`

### Example / Starter Code
```javascript
const express = require('express');
const app = express();
app.use(express.json());

let todos = [];
let nextId = 1;

app.get('/todos', (req, res) => {
  res.json(todos);
});

app.post('/todos', (req, res) => {
  const { title, done = false } = req.body;
  const todo = { id: nextId++, title, done };
  todos.push(todo);
  res.status(201).json(todo);
});

app.get('/todos/:id', (req, res) => {
  const todo = todos.find(t => t.id === parseInt(req.params.id, 10));
  if (!todo) return res.status(404).json({ error: 'Todo not found' });
  res.json(todo);
});

app.put('/todos/:id', (req, res) => {
  const todo = todos.find(t => t.id === parseInt(req.params.id, 10));
  if (!todo) return res.status(404).json({ error: 'Todo not found' });
  const { title, done } = req.body;
  if (title !== undefined) todo.title = title;
  if (done !== undefined) todo.done = done;
  res.json(todo);
});

app.delete('/todos/:id', (req, res) => {
  const index = todos.findIndex(t => t.id === parseInt(req.params.id, 10));
  if (index === -1) return res.status(404).json({ error: 'Todo not found' });
  todos.splice(index, 1);
  res.json({ message: 'Todo deleted' });
});

app.listen(3000, () => console.log('Server on http://127.0.0.1:3000'));
```
**Test with curl:**
```bash
curl http://127.0.0.1:3000/todos
curl -X POST http://127.0.0.1:3000/todos \
  -H "Content-Type: application/json" \
  -d '{"title": "Learn Express"}'
curl http://127.0.0.1:3000/todos/1
curl -X PUT http://127.0.0.1:3000/todos/1 \
  -H "Content-Type: application/json" \
  -d '{"done": true}'
curl -X DELETE http://127.0.0.1:3000/todos/1
```

## Exercise 3: Logging Middleware + 404/Error Handler
**Difficulty:** Basic
### Problem
Add request logging middleware and global error handling:
1. Write a custom middleware that logs `METHOD PATH` for every request.
2. Add a catch-all `404` handler for unknown routes.
3. Add an error-handling middleware (4 args) and a `/boom` route that triggers it.
4. Add a `GET /hello` route that returns a greeting.

### Concepts
- Custom middleware `(req, res, next)`
- `app.use()` ordering
- Error-handling middleware signature `(err, req, res, next)`

### HTTP Method(s)
`GET`

### Example / Starter Code
```javascript
const express = require('express');
const app = express();

// 1. Logging middleware
app.use((req, res, next) => {
  console.log(`${req.method} ${req.path}`);
  next();
});

app.get('/hello', (req, res) => {
  res.json({ message: 'Hello, world!' });
});

app.get('/boom', (req, res, next) => {
  next(new Error('Something went wrong!'));
});

// 2. Catch-all 404 handler
app.use((req, res) => {
  res.status(404).json({ error: 'Route not found' });
});

// 3. Error-handling middleware
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: 'Internal server error' });
});

app.listen(3000, () => console.log('Server on http://127.0.0.1:3000'));
```
**Test with curl:**
```bash
curl http://127.0.0.1:3000/hello
curl http://127.0.0.1:3000/nope   # 404 handler
curl http://127.0.0.1:3000/boom   # 500 handler
```
