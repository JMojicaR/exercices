## Exercise 1: Basic Server + GET Endpoints
**Difficulty:** Basic
**Concepts:** `express()`, `app.get()`, `res.json()`, route params, `app.listen()`

Create an Express server with an in-memory list of movies plus two GET endpoints: all movies and a single movie by `id` (404 if missing).

**Starter code:**
```javascript
const express = require("express");
const app = express();

const movies = [
  { id: 1, title: "Inception", year: 2010 },
  { id: 2, title: "Interstellar", year: 2014 },
  { id: 3, title: "The Matrix", year: 1999 },
];

app.get("/movies", (req, res) => {
  res.json(movies);
});

app.get("/movies/:id", (req, res) => {
  const movie = movies.find((m) => m.id === Number(req.params.id));
  if (!movie) {
    return res.status(404).json({ error: "Movie not found" });
  }
  res.json(movie);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a `GET /movies?year=2010` query filter using `req.query.year`.

## Exercise 2: POST + PUT with Validation
**Difficulty:** Basic
**Concepts:** `express.json()` middleware, `req.body`, `res.status(201)`, validation

Add a `POST /movies` endpoint to create a movie and a `PUT /movies/:id` endpoint to replace one, with basic input validation.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let movies = [
  { id: 1, title: "Inception", year: 2010 },
  { id: 2, title: "Interstellar", year: 2014 },
];

app.post("/movies", (req, res) => {
  const { title, year } = req.body;
  if (!title || typeof title !== "string") {
    return res.status(400).json({ error: "title is required" });
  }
  const newId = movies.length ? Math.max(...movies.map((m) => m.id)) + 1 : 1;
  const movie = { id: newId, title, year };
  movies.push(movie);
  res.status(201).json(movie);
});

app.put("/movies/:id", (req, res) => {
  const movie = movies.find((m) => m.id === Number(req.params.id));
  if (!movie) {
    return res.status(404).json({ error: "Movie not found" });
  }
  const { title, year } = req.body;
  movie.title = title ?? movie.title;
  movie.year = year ?? movie.year;
  res.json(movie);
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Validate that `year` (when provided) is a number between 1888 and 2100, returning 400 otherwise.

## Exercise 3: DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.delete()`, array filtering, custom error middleware, `next()`

Add a `DELETE /movies/:id` endpoint and a global error-handling middleware that returns JSON for unhandled errors and unknown routes.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let movies = [
  { id: 1, title: "Inception", year: 2010 },
  { id: 2, title: "Interstellar", year: 2014 },
];

app.delete("/movies/:id", (req, res) => {
  const id = Number(req.params.id);
  const movie = movies.find((m) => m.id === id);
  if (!movie) {
    return res.status(404).json({ error: "Movie not found" });
  }
  movies = movies.filter((m) => m.id !== id);
  res.json({ message: `Movie ${id} deleted` });
});

// 404 fallback for unknown routes
app.use((req, res) => {
  res.status(404).json({ error: "Route not found" });
});

// Global error handler (4 args)
app.use((err, req, res, next) => {
  console.error(err);
  res.status(500).json({ error: "Internal server error" });
});

app.listen(3000, () => console.log("Server on http://localhost:3000"));
```

**Task:** Add a `GET /movies/:id` endpoint that confirms a deleted movie returns 404 on re-read.
