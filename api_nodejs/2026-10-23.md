## Exercise 1: App Setup + GET Endpoints
**Difficulty:** Basic
**Concepts:** `express()`, routing, `res.json`, middleware basics

Create an Express app with a health-check route and a route that returns all users.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const users = [
    { id: 1, name: "Ana", email: "ana@example.com" },
    { id: 2, name: "Luis", email: "luis@example.com" },
];

app.get("/health", (req, res) => {
    res.json({ status: "ok" });
});

app.get("/users", (req, res) => {
    res.json(users);
});

app.listen(PORT, () => {
    console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a `GET /users/count` endpoint that returns `{ count: users.length }`, and add a simple logging middleware with `app.use()`.

## Exercise 2: POST + express.json() + Validation
**Difficulty:** Basic
**Concepts:** POST, `express.json()`, request body, validation, status codes

Add an endpoint to create a user, parsing JSON and validating required fields.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let users = [];
let nextId = 1;

app.post("/users", (req, res) => {
    const { name, email } = req.body;

    if (!name || !email) {
        return res.status(400).json({ error: "name and email are required" });
    }

    const user = { id: nextId++, name, email };
    users.push(user);
    res.status(201).json(user);
});
```

**Task:** Validate that `email` contains an `@` character (basic check), returning a 400 if not, and add a `GET /users/:id` route.

## Exercise 3: PUT / DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** PUT, DELETE, path params, error-handling middleware

Add update and delete endpoints plus a centralized error handler.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

let users = [];
let nextId = 1;

app.put("/users/:id", (req, res) => {
    const id = Number(req.params.id);
    const user = users.find((u) => u.id === id);
    if (!user) {
        return res.status(404).json({ error: "User not found" });
    }
    const { name, email } = req.body;
    if (name) user.name = name;
    if (email) user.email = email;
    res.json(user);
});

app.delete("/users/:id", (req, res) => {
    const id = Number(req.params.id);
    const index = users.findIndex((u) => u.id === id);
    if (index === -1) {
        return res.status(404).json({ error: "User not found" });
    }
    users.splice(index, 1);
    res.json({ message: "User deleted" });
});

app.use((err, req, res, next) => {
    console.error(err.stack);
    res.status(500).json({ error: "Internal server error" });
});
```

**Task:** Add a `GET /users/:id` route that returns a 404 for a missing user, and test all CRUD operations end-to-end.
