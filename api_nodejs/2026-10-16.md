## Exercise 1: App + GET /notes + GET /health
**Difficulty:** Basic
**Concepts:** Express app, routing, res.json

Create an Express app that serves a list of notes as JSON plus a health-check endpoint.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const notes = [
  { id: 1, title: "Groceries", body: "Milk, eggs, bread" },
  { id: 2, title: "Meeting", body: "Standup at 10am" },
];

app.get("/health", (req, res) => {
  res.json({ status: "ok" });
});

app.get("/notes", (req, res) => {
  res.json(notes);
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a GET /notes/count endpoint that returns the number of notes as JSON.

## Exercise 2: POST /notes + GET /notes/:id + 404 handling
**Difficulty:** Basic
**Concepts:** POST, express.json(), route params, 404

Let clients add notes and fetch one note by id, returning 404 when missing.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const notes = [
  { id: 1, title: "Groceries", body: "Milk, eggs, bread" },
];

app.post("/notes", (req, res) => {
  const note = { id: notes.length + 1, ...req.body };
  notes.push(note);
  res.status(201).json(note);
});

app.get("/notes/:id", (req, res) => {
  const note = notes.find((n) => n.id === Number(req.params.id));
  if (!note) {
    return res.status(404).json({ error: "note not found" });
  }
  res.json(note);
});

app.listen(3000, () => console.log("Server running on http://localhost:3000"));
```

**Task:** Reject POST requests whose body lacks a "title" field with a 400 response.

## Exercise 3: PUT/DELETE + error-handling middleware
**Difficulty:** Basic
**Concepts:** PUT, DELETE, error-handling middleware

Add update and delete endpoints plus a global error-handling middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const notes = [
  { id: 1, title: "Groceries", body: "Milk, eggs, bread" },
];

app.put("/notes/:id", (req, res) => {
  const note = notes.find((n) => n.id === Number(req.params.id));
  if (!note) {
    return res.status(404).json({ error: "note not found" });
  }
  Object.assign(note, req.body);
  res.json(note);
});

app.delete("/notes/:id", (req, res) => {
  const index = notes.findIndex((n) => n.id === Number(req.params.id));
  if (index === -1) {
    return res.status(404).json({ error: "note not found" });
  }
  notes.splice(index, 1);
  res.json({ message: "deleted" });
});

// Error-handling middleware (must have 4 arguments)
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(400).json({ error: "bad request" });
});

app.listen(3000, () => console.log("Server running on http://localhost:3000"));
```

**Task:** Add a catch-all 404 middleware for unknown routes that returns JSON instead of the default HTML.
