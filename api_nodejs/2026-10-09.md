# Node.js / Express — API Exercises (2026-10-09)
Sources: Express official docs, MDN, GeeksforGeeks

**Setup:** `npm init -y && npm install express`, run with `node app.js`. Theme: a Posts API.

## Exercise 1: App + GET /posts
**Difficulty:** Basic
**Concepts:** Express app, routing, res.json

Create an Express app that serves a list of posts as JSON.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const posts = [
  { id: 1, title: "Hello World", author: "ada" },
  { id: 2, title: "Express Basics", author: "alan" },
];

app.get("/posts", (req, res) => {
  res.json(posts);
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

## Exercise 2: POST /posts + express.json() + validation
**Difficulty:** Basic
**Concepts:** POST, JSON body parsing, validation

Let clients create posts, validating that a title is provided.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());

const posts = [
  { id: 1, title: "Hello World", author: "ada" },
];

app.get("/posts", (req, res) => {
  res.json(posts);
});

app.post("/posts", (req, res) => {
  const { title, author } = req.body;
  if (!title) {
    return res.status(400).json({ error: "title is required" });
  }
  const post = { id: posts.length + 1, title, author: author || "anonymous" };
  posts.push(post);
  res.status(201).json(post);
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add GET /posts/:id that returns a single post or a 404 JSON error.

## Exercise 3: PUT/DELETE + error-handling middleware
**Difficulty:** Basic
**Concepts:** PUT, DELETE, middleware, error handling

Add update and delete endpoints plus a global error-handling middleware.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());

const posts = [
  { id: 1, title: "Hello World", author: "ada" },
];

app.get("/posts", (req, res) => {
  res.json(posts);
});

app.put("/posts/:id", (req, res) => {
  const post = posts.find((p) => p.id === Number(req.params.id));
  if (!post) {
    return res.status(404).json({ error: "post not found" });
  }
  const { title, author } = req.body;
  if (title) post.title = title;
  if (author) post.author = author;
  res.json(post);
});

app.delete("/posts/:id", (req, res) => {
  const index = posts.findIndex((p) => p.id === Number(req.params.id));
  if (index === -1) {
    return res.status(404).json({ error: "post not found" });
  }
  posts.splice(index, 1);
  res.json({ message: "deleted" });
});

// Global error-handling middleware (must have 4 args)
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: "internal server error" });
});

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a 404 catch-all middleware for unknown routes (`app.use((req, res) => ...)`).
