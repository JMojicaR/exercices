## Exercise 1: Basic Server + GET Endpoints
**Difficulty:** Basic
**Concepts:** `express`, `app.get`, `res.json`, route parameters, `res.status`

Create an Express app with an in-memory array of events and two GET endpoints: one returning all events and one returning a single event by `id` (404 if missing).

**Starter code:**
```javascript
const express = require('express');
const app = express();
app.use(express.json());

const events = [
  { id: 1, name: 'Tech Meetup', date: '2026-12-01', capacity: 100 },
  { id: 2, name: 'Music Festival', date: '2026-12-15', capacity: 5000 },
  { id: 3, name: 'Cooking Workshop', date: '2026-11-20', capacity: 30 },
];

app.get('/events', (req, res) => {
  res.json(events);
});

app.get('/events/:id', (req, res) => {
  const event = events.find(e => e.id === Number(req.params.id));
  if (!event) return res.status(404).json({ error: 'Event not found' });
  res.json(event);
});

app.listen(3000, () => console.log('Server running on port 3000'));
```

**Task:** Add a `GET /events?minCapacity=50` query filter using `req.query.minCapacity`.

## Exercise 2: POST + PUT (Create & Update with Validation)
**Difficulty:** Basic
**Concepts:** `req.body`, `app.post`, `app.put`, returning 201, basic validation

Add a `POST /events` endpoint to create an event and a `PUT /events/:id` endpoint to replace one, both with simple validation.

**Starter code:**
```javascript
const express = require('express');
const app = express();
app.use(express.json());

const events = [
  { id: 1, name: 'Tech Meetup', date: '2026-12-01', capacity: 100 },
  { id: 2, name: 'Music Festival', date: '2026-12-15', capacity: 5000 },
];

app.post('/events', (req, res) => {
  const { name, date, capacity } = req.body;
  if (!name || !date) {
    return res.status(400).json({ error: 'name and date are required' });
  }
  const newId = events.length ? Math.max(...events.map(e => e.id)) + 1 : 1;
  const event = { id: newId, name, date, capacity };
  events.push(event);
  res.status(201).json(event);
});

app.put('/events/:id', (req, res) => {
  const event = events.find(e => e.id === Number(req.params.id));
  if (!event) return res.status(404).json({ error: 'Event not found' });
  const { name, date, capacity } = req.body;
  if (name !== undefined) event.name = name;
  if (date !== undefined) event.date = date;
  if (capacity !== undefined) event.capacity = capacity;
  res.json(event);
});

app.listen(3000, () => console.log('Server running on port 3000'));
```

**Task:** Add validation that `capacity` must be a positive number, returning `400` if it is missing, negative, or not a number.

## Exercise 3: DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.delete`, middleware `next(err)`, centralized error handler

Add a `DELETE /events/:id` endpoint and a centralized error-handling middleware that returns JSON for unknown routes and server errors.

**Starter code:**
```javascript
const express = require('express');
const app = express();
app.use(express.json());

const events = [
  { id: 1, name: 'Tech Meetup', date: '2026-12-01', capacity: 100 },
  { id: 2, name: 'Music Festival', date: '2026-12-15', capacity: 5000 },
];

app.delete('/events/:id', (req, res) => {
  const index = events.findIndex(e => e.id === Number(req.params.id));
  if (index === -1) return res.status(404).json({ error: 'Event not found' });
  events.splice(index, 1);
  res.json({ message: `Event ${req.params.id} deleted` });
});

// 404 for unknown routes
app.use((req, res) => {
  res.status(404).json({ error: 'Route not found' });
});

// Centralized error handler (must have 4 parameters)
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: 'Internal server error' });
});

app.listen(3000, () => console.log('Server running on port 3000'));
```

**Task:** Explain why the error-handling middleware must be declared with **four** parameters (`err, req, res, next`) for Express to recognize it as an error handler, and test it by throwing an error inside a route.
