## Exercise 1: App Setup + GET Endpoints
**Difficulty:** Basic
**Concepts:** `express()`, routing, `res.json()`, `app.listen()`

Create an Express app with a health-check route and a route that returns all products.

**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

const products = [
    { id: 1, name: "Laptop", price: 999 },
    { id: 2, name: "Mouse", price: 25 },
];

app.get("/health", (req, res) => {
    res.json({ status: "ok" });
});

app.get("/products", (req, res) => {
    res.json(products);
});

app.listen(PORT, () => {
    console.log(`Server running on http://localhost:${PORT}`);
});
```

**Task:** Add a `GET /products/count` endpoint that returns `{ count: products.length }`, and a `GET /products/search?name=...` filter route.

## Exercise 2: POST + GET by ID (404 handling)
**Difficulty:** Basic
**Concepts:** `app.post`, `express.json()`, path params, `res.status()`

Add endpoints to create a product and fetch a single product by ID.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const products = [];
let nextId = 1;

app.post("/products", (req, res) => {
    const { name, price } = req.body;
    if (!name) {
        return res.status(400).json({ error: "name is required" });
    }
    const product = { id: nextId++, name, price: price ?? 0 };
    products.push(product);
    res.status(201).json(product);
});

app.get("/products/:id", (req, res) => {
    const product = products.find((p) => p.id === Number(req.params.id));
    if (!product) {
        return res.status(404).json({ error: "Product not found" });
    }
    res.json(product);
});
```

**Task:** Validate that `price` is a non-negative number; if not, return a 400 error with a descriptive message.

## Exercise 3: PUT + DELETE + Error-Handling Middleware
**Difficulty:** Basic
**Concepts:** `app.put`, `app.delete`, 404 middleware, error middleware

Add endpoints to update and delete a product, plus centralized error handling.

**Starter code:**
```javascript
const express = require("express");
const app = express();
app.use(express.json());

const products = [];

app.put("/products/:id", (req, res) => {
    const product = products.find((p) => p.id === Number(req.params.id));
    if (!product) {
        return res.status(404).json({ error: "Product not found" });
    }
    product.name = req.body.name ?? product.name;
    product.price = req.body.price ?? product.price;
    res.json(product);
});

app.delete("/products/:id", (req, res) => {
    const index = products.findIndex((p) => p.id === Number(req.params.id));
    if (index === -1) {
        return res.status(404).json({ error: "Product not found" });
    }
    products.splice(index, 1);
    res.json({ message: "Product deleted" });
});

// 404 middleware (must be last)
app.use((req, res) => {
    res.status(404).json({ error: "Route not found" });
});

// error-handling middleware (4 args)
app.use((err, req, res, next) => {
    console.error(err.message);
    res.status(500).json({ error: "Internal server error" });
});
```

**Task:** Add a 400 handler for malformed JSON, and log every request with a simple logging middleware (`req.method` + `req.url`).
