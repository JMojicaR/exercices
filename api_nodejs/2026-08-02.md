## Exercise 1: Basic CRUD Routes with Express
- **Difficulty**: Basic
- **Concepts**: Express app setup, route handlers, req.params, req.query, req.body, res.json, res.status, in-memory storage
- **Description**: Build a REST API for managing notes. Each note has an `id`, `title`, `content`, and `createdAt`. Use Express Router for clean route organization. Support GET (all/single), POST, PUT, and DELETE. Parse JSON bodies with `express.json()` middleware.

### Example/Starter Code
```javascript
const express = require('express');
const app = express();
const PORT = 3000;

// Middleware
app.use(express.json());

// In-memory storage
let notes = [];
let nextId = 1;

// GET /notes — list all notes
app.get('/notes', (req, res) => {
  // TODO: Support optional query param ?search=keyword to filter by title
  const { search } = req.query;
  if (search) {
    return res.json(notes.filter(n =>
      n.title.toLowerCase().includes(search.toLowerCase())
    ));
  }
  res.json(notes);
});

// GET /notes/:id — get a single note
app.get('/notes/:id', (req, res) => {
  const id = parseInt(req.params.id, 10);
  const note = notes.find(n => n.id === id);
  if (!note) {
    return res.status(404).json({ error: 'Note not found' });
  }
  res.json(note);
});

// POST /notes — create a new note
app.post('/notes', (req, res) => {
  const { title, content } = req.body;

  // TODO: Validate that title is present and non-empty
  // TODO: Validate that content is present
  if (!title || !title.trim()) {
    return res.status(400).json({ error: 'Title is required and must not be empty' });
  }
  if (!content) {
    return res.status(400).json({ error: 'Content is required' });
  }

  const note = {
    id: nextId++,
    title: title.trim(),
    content,
    createdAt: new Date().toISOString()
  };
  notes.push(note);
  res.status(201).json(note);
});

// PUT /notes/:id — update a note
app.put('/notes/:id', (req, res) => {
  const id = parseInt(req.params.id, 10);
  const index = notes.findIndex(n => n.id === id);
  if (index === -1) {
    return res.status(404).json({ error: 'Note not found' });
  }

  // TODO: Update only the fields that are provided
  const { title, content } = req.body;
  if (title !== undefined) {
    if (!title.trim()) {
      return res.status(400).json({ error: 'Title must not be empty' });
    }
    notes[index].title = title.trim();
  }
  if (content !== undefined) {
    notes[index].content = content;
  }
  res.json(notes[index]);
});

// DELETE /notes/:id — delete a note
app.delete('/notes/:id', (req, res) => {
  const id = parseInt(req.params.id, 10);
  const index = notes.findIndex(n => n.id === id);
  if (index === -1) {
    return res.status(404).json({ error: 'Note not found' });
  }
  notes.splice(index, 1);
  res.status(204).send();
});

// TODO: Test with curl:

// curl -X PUT http://localhost:3000/notes/1 -H "Content-Type: application/json" -d '{"title":"Updated"}'
// curl -X DELETE http://localhost:3000/notes/1

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

## Exercise 2: Middleware and Error Handling
- **Difficulty**: Basic
- **Concepts**: Custom middleware, error-handling middleware (4-argument signature), async error wrapping, request logging, CORS
- **Description**: Enhance the notes API with custom middleware: request logger, error handler, CORS support, and an async wrapper to catch errors from async route handlers. Create a centralized error class and demonstrate the Express error-handling pipeline.

### Example/Starter Code
```javascript
const express = require('express');
const cors = require('cors');
const app = express();
const PORT = 3000;

// --- Custom Error Class ---
class AppError extends Error {
  constructor(message, statusCode) {
    super(message);
    this.statusCode = statusCode;
    this.isOperational = true;
  }
}

// --- Middleware ---
app.use(express.json());
app.use(cors());  // Enable CORS for all routes

// Request logger middleware
app.use((req, res, next) => {
  const start = Date.now();
  const { method, url } = req;

  // Log when response finishes
  res.on('finish', () => {
    const duration = Date.now() - start;
    console.log(`${method} ${url} ${res.statusCode} - ${duration}ms`);
  });

  next();
});

// --- Async Error Wrapper ---
// Wraps async route handlers so thrown errors get passed to next()
const asyncHandler = (fn) => (req, res, next) => {
  Promise.resolve(fn(req, res, next)).catch(next);
};

// --- Routes (using asyncHandler) ---
let notes = [];
let nextId = 1;

app.get('/notes', asyncHandler(async (req, res) => {
  // Simulate an async DB query
  const result = await new Promise(resolve =>
    setTimeout(() => resolve(notes), 50)
  );
  res.json(result);
}));

app.post('/notes', asyncHandler(async (req, res) => {
  const { title, content } = req.body;

  // TODO: Use AppError for validation failures
  if (!title || !title.trim()) {
    throw new AppError('Title is required', 400);
  }

  const note = { id: nextId++, title: title.trim(), content };
  notes.push(note);
  res.status(201).json(note);
}));

// --- 404 Handler (for unmatched routes) ---
app.use((req, res, next) => {
  next(new AppError(`Route ${req.method} ${req.url} not found`, 404));
});

// --- Global Error Handling Middleware (4 params = error handler) ---
app.use((err, req, res, next) => {
  const statusCode = err.statusCode || 500;
  const message = err.isOperational ? err.message : 'Internal server error';

  // Log non-operational (unexpected) errors
  if (!err.isOperational) {
    console.error('UNEXPECTED ERROR:', err);
  }

  res.status(statusCode).json({
    error: message,
    ...(process.env.NODE_ENV === 'development' && { stack: err.stack })
  });
});

// TODO: Test error handling: POST with missing title, GET a non-existent route
// TODO: Create a rate-limiting middleware

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```

## Exercise 3: Request Validation and Structured API Design
- **Difficulty**: Basic
- **Concepts**: Input validation (express-validator or Joi), route organization with Express Router, request sanitization, structured responses
- **Description**: Refactor the notes API using Express Router for modular route files. Add input validation with `express-validator` to validate and sanitize request bodies. Create a consistent response format (wrapper object with `success`, `data`, `error` fields). Handle validation errors globally.

### Example/Starter Code
```javascript
const express = require('express');
const { body, param, query, validationResult } = require('express-validator');

const app = express();
const PORT = 3000;

app.use(express.json());

// --- Consistent Response Helper ---
function successResponse(res, data, statusCode = 200) {
  return res.status(statusCode).json({ success: true, data });
}

function errorResponse(res, message, statusCode = 400, errors = null) {
  return res.status(statusCode).json({ success: false, error: message, errors });
}

// --- Validation Middleware ---
function validate(req, res, next) {
  const errors = validationResult(req);
  if (!errors.isEmpty()) {
    return errorResponse(res, 'Validation failed', 422, errors.array());
  }
  next();
}

// --- Route Handlers with Validation Chains ---
let notes = [];
let nextId = 1;

// GET /notes?sort=asc|desc&search=keyword
app.get('/notes', [
  query('search').optional().isString().trim().escape(),
  query('sort').optional().isIn(['asc', 'desc']),
  validate,
], (req, res) => {
  let result = [...notes];
  const { search, sort } = req.query;

  if (search) {
    result = result.filter(n =>
      n.title.toLowerCase().includes(search.toLowerCase())
    );
  }

  if (sort === 'asc') {
    result.sort((a, b) => a.title.localeCompare(b.title));
  } else if (sort === 'desc') {
    result.sort((a, b) => b.title.localeCompare(a.title));
  }

  return successResponse(res, result);
});

// GET /notes/:id
app.get('/notes/:id', [
  param('id').isInt({ min: 1 }).withMessage('ID must be a positive integer'),
  validate,
], (req, res) => {
  const id = parseInt(req.params.id, 10);
  const note = notes.find(n => n.id === id);
  if (!note) {
    return errorResponse(res, 'Note not found', 404);
  }
  return successResponse(res, note);
});

// POST /notes
app.post('/notes', [
  body('title')
    .trim()
    .notEmpty().withMessage('Title is required')
    .isLength({ max: 200 }).withMessage('Title must be at most 200 characters')
    .escape(),
  body('content')
    .trim()
    .notEmpty().withMessage('Content is required'),
  validate,
], (req, res) => {
  const { title, content } = req.body;

  const note = {
    id: nextId++,
    title,
    content,
    createdAt: new Date().toISOString()
  };
  notes.push(note);
  return successResponse(res, note, 201);
});

// PUT /notes/:id
app.put('/notes/:id', [
  param('id').isInt({ min: 1 }),
  body('title').optional().trim().notEmpty().isLength({ max: 200 }).escape(),
  body('content').optional().trim().notEmpty(),
  validate,
], (req, res) => {
  const id = parseInt(req.params.id, 10);
  const index = notes.findIndex(n => n.id === id);
  if (index === -1) {
    return errorResponse(res, 'Note not found', 404);
  }

  const { title, content } = req.body;
  if (title !== undefined) notes[index].title = title;
  if (content !== undefined) notes[index].content = content;

  return successResponse(res, notes[index]);
});

// DELETE /notes/:id
app.delete('/notes/:id', [
  param('id').isInt({ min: 1 }),
  validate,
], (req, res) => {
  const id = parseInt(req.params.id, 10);
  const index = notes.findIndex(n => n.id === id);
  if (index === -1) {
    return errorResponse(res, 'Note not found', 404);
  }
  notes.splice(index, 1);
  return successResponse(res, null, 204);
});

// TODO: Refactor routes into a separate `routes/notes.js` file using express.Router()
// TODO: Add sanitization for all string inputs
// TODO: Test validation: POST with missing title, excessively long title, non-integer ID

app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
```
