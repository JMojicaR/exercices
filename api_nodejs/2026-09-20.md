## 1. Hello World + Route & Query Parameters
**Description:** Create an Express app with a `GET /` root route, a `GET /greet/:name` route using a route parameter, and a `GET /add` route reading query parameters (`?a=2&b=3`).
**Difficulty:** Basic
**Concepts:** `express()`, `app.get`, `req.params`, `req.query`, `res.json`, `app.listen`.
**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.get("/", (req, res) => {
  res.json({ message: "Hello, World!" });
});

app.get("/greet/:name", (req, res) => {
  res.json({ message: `Hello, ${req.params.name}!` });
});

app.get("/add", (req, res) => {
  const a = parseInt(req.query.a || "0");
  const b = parseInt(req.query.b || "0");
  res.json({ result: a + b });
});

app.listen(PORT, () => {
  console.log(`Server running at http://localhost:${PORT}`);
});
```

## 2. In-Memory CRUD API (Products)
**Description:** Build a full CRUD API for "products" stored in an in-memory array. Implement `GET /products`, `GET /products/:id`, `POST /products`, `PUT /products/:id`, and `DELETE /products/:id`. Use `express.json()` to parse request bodies.
**Difficulty:** Basic
**Concepts:** `express.json()`, HTTP methods, `req.body`, status codes, array methods.
**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());

let products = [];
let nextId = 1;

app.get("/products", (req, res) => {
  res.json(products);
});

app.get("/products/:id", (req, res) => {
  const id = parseInt(req.params.id);
  const product = products.find((p) => p.id === id);
  if (!product) return res.status(404).json({ error: "Not found" });
  res.json(product);
});

app.post("/products", (req, res) => {
  const { name, price } = req.body;
  const product = { id: nextId++, name, price };
  products.push(product);
  res.status(201).json(product);
});

app.put("/products/:id", (req, res) => {
  const id = parseInt(req.params.id);
  const product = products.find((p) => p.id === id);
  if (!product) return res.status(404).json({ error: "Not found" });
  product.name = req.body.name ?? product.name;
  product.price = req.body.price ?? product.price;
  res.json(product);
});

app.delete("/products/:id", (req, res) => {
  const id = parseInt(req.params.id);
  const before = products.length;
  products = products.filter((p) => p.id !== id);
  if (products.length === before) {
    return res.status(404).json({ error: "Not found" });
  }
  res.status(204).send();
});

app.listen(PORT, () => console.log(`Server running at http://localhost:${PORT}`));
```

## 3. Middleware & Error Handling
**Description:** Add a custom logging middleware that runs on every request, and an error-handling middleware (4-argument signature) that returns JSON for unhandled errors. Also validate POST input and return 400 on invalid data.
**Difficulty:** Basic
**Concepts:** `app.use` middleware, `next()`, error-handling middleware `(err, req, res, next)`, status codes.
**Starter code:**
```javascript
const express = require("express");
const app = express();
const PORT = 3000;

app.use(express.json());

// Logging middleware
app.use((req, res, next) => {
  console.log(`${req.method} ${req.path}`);
  next();
});

const items = [];

app.post("/items", (req, res) => {
  const { name } = req.body;
  if (!name) {
    return res.status(400).json({ error: "Field 'name' is required" });
  }
  items.push(name);
  res.status(201).json({ added: name });
});

// Route that triggers an error
app.get("/boom", (req, res, next) => {
  next(new Error("Something went wrong"));
});

// Error-handling middleware (must have 4 params)
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: err.message });
});

app.listen(PORT, () => console.log(`Server running at http://localhost:${PORT}`));
```
