## Exercise 5: Higher-Order Functions

**Title:** Map and Filter with Higher-Order Functions

**Description:**

**Concepts Practiced:** Higher-order functions, funs (anonymous functions), recursion, function passing

**Difficulty:** Basic

**Starter Code:**

-module(hof).
-export([my_map/2, my_filter/2, double/1, is_even/1]).

%% Apply Fun to each element of the list
my_map(_, []) -> [];
my_map(Fun, [H|T]) ->
    %% Apply Fun to H, recurse on tail
    undefined.

%% Keep only elements where Pred returns true
my_filter(_, []) -> [];
my_filter(Pred, [H|T]) ->
    case Pred(H) of
        true ->
            %% Keep H, recurse on T
            undefined;
        false ->
            %% Skip H, recurse on T
            my_filter(Pred, T)
    end.

%% Helper: double a number
double(X) -> X * 2.

%% Helper: check if a number is even
is_even(X) -> X rem 2 =:= 0.

%% Example usage:
%% 1> c(hof).
%% 2> hof:my_map(fun hof:double/1, [1,2,3,4,5]).
