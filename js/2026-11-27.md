## Exercise 1: Define a Node and Build a Small Tree
**Difficulty:** Basic
**Concepts:** ES6 `class`, `constructor`, object references

**Starter code:**
```javascript
class Node {
  constructor(value) {
    this.value = value;
    this.left = null;
    this.right = null;
  }
}

//         8
//        / \
//       3   10
const root = new Node(8);
root.left = new Node(3);
root.right = new Node(10);
```

**Task:** Extend the tree to match the shape below, then log `root.left.right.value`:
```
       8
      / \
     3   10
    / \    \
   1   6    14
```
(Expected output: `6`.)

## Exercise 2: Insert a Value into the BST
**Difficulty:** Basic
**Concepts:** recursive function, `null` base case, comparing values

**Starter code:**
```javascript
function insert(node, value) {
  if (node === null) return new Node(value);
  if (value < node.value) node.left = insert(node.left, value);
  else node.right = insert(node.right, value);
  return node;
}

let root = null;
for (const v of [8, 3, 10, 1, 6]) {
  root = insert(root, v);
}
```

**Task:** Insert `14`, `4`, and `7`, then write a small recursive function that prints `value (left, right)` for every node to verify the structure.

## Exercise 3: Search for a Value
**Difficulty:** Basic
**Concepts:** recursive search, pruning half the tree each call, `true`/`false` returns

**Starter code:**
```javascript
function contains(node, value) {
  if (node === null) return false;
  if (node.value === value) return true;
  if (value < node.value) return contains(node.left, value);
  return contains(node.right, value);
}

let root = null;
for (const v of [8, 3, 10, 1, 6, 14, 4, 7]) {
  root = insert(root, v);
}

console.log(contains(root, 6));   // true
console.log(contains(root, 12));  // false
```

**Task:** Modify `contains` to return the depth (steps from the root) where the value sits, or `-1` if absent.

## Exercise 4: In-Order Traversal (Sorted Output)
**Difficulty:** Basic
**Concepts:** left → current → right recursion, building an array result

**Starter code:**
```javascript
function inorder(node, out = []) {
  if (node === null) return out;
  inorder(node.left, out);
  out.push(node.value);
  inorder(node.right, out);
  return out;
}

let root = null;
for (const v of [8, 3, 10, 1, 6, 14, 4, 7]) {
  root = insert(root, v);
}

console.log(inorder(root));  // [1, 3, 4, 6, 7, 8, 10, 14]
```

**Task:** Add a `preorder(node, out = [])` function (current → left → right) and log both. Confirm `inorder` is sorted and `preorder` begins with `8`.

## Exercise 5: Find the Minimum and Maximum
**Difficulty:** Basic
**Concepts:** following `left`/`right` chains, `while` loop over references

**Starter code:**
```javascript
function minValue(node) {
  while (node.left !== null) {
    node = node.left;
  }
  return node.value;
}

function maxValue(node) {
  while (node.right !== null) {
    node = node.right;
  }
  return node.value;
}

let root = null;
for (const v of [8, 3, 10, 1, 6, 14, 4, 7]) {
  root = insert(root, v);
}

console.log(minValue(root));  // 1
console.log(maxValue(root));  // 14
```

**Task:** Rewrite both functions using recursion instead of a `while` loop.
