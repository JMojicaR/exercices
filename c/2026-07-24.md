```c
    if (b == 0) {
        printf("Error: Division by zero!\n");
        return 0;

    return a / b;
}

    float num1, num2;
    char op;

    printf("Enter expression (e.g., 5 + 3): ");
    scanf("%f %c %f", &num1, &op, &num2);

    // Your switch statement here

    return 0;
}
```

**Expected Learning:** Function declaration and calling, `switch/case`, `float` data type, error handling, organized code structure.
