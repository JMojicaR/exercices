**Starter Code:**

    // Set *min to the smallest value and *max to the largest value in arr
}

int main() {
    int arr[] = {7, 2, 9, 1, 5, 3, 8, 4, 6};
    int size = sizeof(arr) / sizeof(arr[0]);
    int min, max;

    find_min_max(arr, size, &min, &max);
    printf("Minimum: %d\n", min);  // Expected: 1
    printf("Maximum: %d\n", max);  // Expected: 9

    // Test with single-element array
    int single[] = {42};
    find_min_max(single, 1, &min, &max);
    printf("Single element - Min: %d, Max: %d\n", min, max);  // Both 42

}
