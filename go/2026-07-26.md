## Exercise 1: Hello World & Packages
**Difficulty:** ★☆☆☆☆  
**Concepts:** package main, imports, fmt, variables, := syntax  

**Description:**  
Write a Go program that demonstrates variable declaration (both `var` and `:=`), multiple return values, and formatted output. Create a helper function that returns a greeting.

**Example Starter Code:**
```go
package main

import (
    "fmt"
    "strings"
)

// Function with multiple return values
func getGreeting(name string, age int) (string, int) {
    greeting := fmt.Sprintf("Hello, %s! You are %d years old.", name, age)
    return greeting, 2026 - age
}

func main() {
    // Variable declarations
    var language string = "Go"
    version := "1.21"  // Short declaration
    
    fmt.Println("=== Welcome to", language, version, "===\n")
    
    // User profile
    name := "Eve"
    age := 28
    isLearning := true
    
    fmt.Printf("Name: %s\n", name)
    fmt.Printf("Age: %d\n", age)
    fmt.Printf("Learning Go: %t\n\n", isLearning)
    
    // Multiple return values
    message, birthYear := getGreeting(name, age)
    fmt.Println(message)
    fmt.Printf("Born around: %d\n", birthYear)
    
    // String manipulation
    upperName := strings.ToUpper(name)
    fmt.Printf("\nUppercase name: %s\n", upperName)
    
    // Type information
    fmt.Printf("\nType of age: %T\n", age)
    fmt.Printf("Type of version: %T\n", version)
}
```

**Expected Output:**
```
=== Welcome to Go 1.21 ===

Name: Eve
Age: 28
Learning Go: true

Hello, Eve! You are 28 years old.
Born around: 1998

Uppercase name: EVE

Type of age: int
Type of version: string
```

## Exercise 2: Goroutines & Channels
**Difficulty:** ★★★☆☆  
**Concepts:** Goroutines, channels, concurrency, sync.WaitGroup  

**Description:**  
Use goroutines to run tasks concurrently. Launch multiple goroutines that simulate work, communicate results via channels, and synchronize with a WaitGroup.

**Example Starter Code:**
```go
package main

import (
    "fmt"
    "sync"
    "time"
)

func worker(id int, wg *sync.WaitGroup, results chan<- string) {
    defer wg.Done()
    
    // Simulate work
    duration := time.Duration(id*100) * time.Millisecond
    time.Sleep(duration)
    
    result := fmt.Sprintf("Worker %d finished after %v", id, duration)
    results <- result
}

func main() {
    fmt.Println("=== Concurrent Workers ===\n")
    
    var wg sync.WaitGroup
    results := make(chan string, 5)
    
    start := time.Now()
    
    // Launch 5 workers
    for i := 1; i <= 5; i++ {
        wg.Add(1)
        go worker(i, &wg, results)
    }
    
    // Wait for all workers in a separate goroutine
    go func() {
        wg.Wait()
        close(results)
    }()
    
    // Collect results
    for result := range results {
        fmt.Println(result)
    }
    
    elapsed := time.Since(start)
    fmt.Printf("\nAll workers completed in %v\n", elapsed)
}
```

## Exercise 3: Structs & Methods
**Difficulty:** ★★★☆☆  
**Concepts:** Structs, methods (value vs pointer receivers), interfaces  

**Description:**  
Define an `Employee` struct and implement methods. Create an interface `Payable` and have `Employee` satisfy it. Demonstrate value vs pointer receivers.

**Example Starter Code:**
```go
package main

import "fmt"

// Define an interface
type Payable interface {
    CalculatePay() float64
    GetInfo() string
}

// Employee struct
type Employee struct {
    ID        int
    FirstName string
    LastName  string
    HourlyRate float64
    HoursWorked float64
}

// Pointer receiver method (modifies the struct)
func (e *Employee) AddHours(hours float64) {
    e.HoursWorked += hours
}

// Value receiver method (read-only)
func (e Employee) CalculatePay() float64 {
    return e.HourlyRate * e.HoursWorked
}

func (e Employee) GetInfo() string {
    return fmt.Sprintf("Employee #%d: %s %s | Rate: $%.2f/hr | Hours: %.1f",
        e.ID, e.FirstName, e.LastName, e.HourlyRate, e.HoursWorked)
}

// Constructor function
func NewEmployee(id int, firstName, lastName string, rate float64) *Employee {
    return &Employee{
        ID:         id,
        FirstName:  firstName,
        LastName:   lastName,
        HourlyRate: rate,
        HoursWorked: 0,
    }
}

func printPayroll(employees []Payable) {
    fmt.Println("\n=== Payroll ===")
    for _, emp := range employees {
        fmt.Println(emp.GetInfo())
        fmt.Printf("  Weekly Pay: $%.2f\n\n", emp.CalculatePay())
    }
}

func main() {
    emp1 := NewEmployee(1, "Frank", "Castle", 25.50)
    emp2 := NewEmployee(2, "Karen", "Page", 30.00)
    
    // Add hours worked
    emp1.AddHours(40)
    emp1.AddHours(5)  // Overtime
    emp2.AddHours(35)
    
    fmt.Println(emp1.GetInfo())
    fmt.Printf("Pay: $%.2f\n\n", emp1.CalculatePay())
    
    // Interface usage
    employees := []Payable{emp1, emp2}
    printPayroll(employees)
}
```

## Exercise 4: Slice Operations
**Difficulty:** ★★☆☆☆  
**Concepts:** Slices, append, copy, range, make, capacity  

**Description:**  
Work with slices: create, append, copy, and iterate. Demonstrate slice capacity behavior and write functions that manipulate slices.

**Example Starter Code:**
```go
package main

import (
    "fmt"
    "sort"
)

func findMinMax(numbers []int) (int, int) {
    if len(numbers) == 0 {
        return 0, 0
    }
    min, max := numbers[0], numbers[0]
    for _, n := range numbers {
        if n < min {
            min = n
        }
        if n > max {
            max = n
        }
    }
    return min, max
}

func filterEven(numbers []int) []int {
    var evens []int
    for _, n := range numbers {
        if n%2 == 0 {
            evens = append(evens, n)
        }
    }
    return evens
}

func main() {
    // Create slices
    numbers := []int{15, 3, 22, 8, 45, 12, 7, 31, 9}
    fmt.Println("Original:", numbers)
    
    // Append
    numbers = append(numbers, 50)
    fmt.Println("After append:", numbers)
    fmt.Printf("Length: %d, Capacity: %d\n\n", len(numbers), cap(numbers))
    
    // Copy
    copied := make([]int, len(numbers))
    copy(copied, numbers)
    fmt.Println("Copied:", copied)
    
    // Slice operations
    fmt.Println("First 3:", numbers[:3])
    fmt.Println("Last 3:", numbers[len(numbers)-3:])
    
    // Sort
    sorted := make([]int, len(numbers))
    copy(sorted, numbers)
    sort.Ints(sorted)
    fmt.Println("\nSorted:", sorted)
    
    // Min/Max
    min, max := findMinMax(numbers)
    fmt.Printf("Min: %d, Max: %d\n", min, max)
    
    // Filter evens
    evens := filterEven(numbers)
    fmt.Println("Evens:", evens)
    
    // Sum
    sum := 0
    for _, n := range numbers {
        sum += n
    }
    fmt.Printf("Sum: %d, Average: %.2f\n", sum, float64(sum)/float64(len(numbers)))
}
```

## Exercise 5: Error Handling
**Difficulty:** ★★★☆☆  
**Concepts:** error interface, custom errors, panic/recover, defer  

**Description:**  
Write functions that return errors. Create a custom error type. Use `defer` for cleanup. Handle division by zero and invalid input gracefully.

**Example Starter Code:**
```go
package main

import (
    "errors"
    "fmt"
)

// Custom error type
type ValidationError struct {
    Field string
    Value interface{}
    Reason string
}

func (e *ValidationError) Error() string {
    return fmt.Sprintf("Validation failed for '%s': value '%v' - %s",
        e.Field, e.Value, e.Reason)
}

// Safe division with error
func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("division by zero")
    }
    return a / b, nil
}

// Validated age input
func validateAge(age int) error {
    if age < 0 {
        return &ValidationError{
            Field:  "age",
            Value:  age,
            Reason: "age cannot be negative",
        }
    }
    if age > 150 {
        return &ValidationError{
            Field:  "age",
            Value:  age,
            Reason: "age seems unreasonably high",
        }
    }
    return nil
}

// Function with defer
func processFile(filename string) {
    fmt.Printf("Opening %s...\n", filename)
    defer fmt.Printf("Closing %s (deferred)\n", filename)
    
    // Simulate work
    fmt.Println("Processing...")
    // If a panic occurs, defer still runs
}

func main() {
    fmt.Println("=== Error Handling ===\n")
    
    // 1. Basic error handling
    testCases := [][]float64{{10, 2}, {5, 0}, {15, 3}}
    
    for _, tc := range testCases {
        result, err := divide(tc[0], tc[1])
        if err != nil {
            fmt.Printf("%.0f / %.0f -> Error: %v\n", tc[0], tc[1], err)
        } else {
            fmt.Printf("%.0f / %.0f = %.2f\n", tc[0], tc[1], result)
        }
    }
    
    // 2. Custom error type
    fmt.Println()
    ages := []int{25, -5, 200}
    for _, age := range ages {
        err := validateAge(age)
        if err != nil {
            fmt.Printf("Age %d: %v\n", age, err)
        } else {
            fmt.Printf("Age %d: Valid ✓\n", age)
        }
    }
    
    // 3. Defer demonstration
    fmt.Println()
    processFile("data.txt")
    
    // 4. Panic and recover
    fmt.Println()
    safeFunction()
    fmt.Println("Program continues after recover!")
}

func safeFunction() {
    defer func() {
        if r := recover(); r != nil {
            fmt.Printf("Recovered from panic: %v\n", r)
        }
    }()
    fmt.Println("About to panic...")
    panic("something went wrong!")
}
```
